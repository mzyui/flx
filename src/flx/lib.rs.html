3//! Fast proxy scraper and validator.
4//!
5//! The [`Flx`] builder mirrors the CLI defaults: scrape the built-in
6//! providers or load candidates from files, validate them against online
7//! judges, then filter, sort, and export the survivors.
8//!
9//! # Examples
10//!
11//! ```no_run
12//! use flx::{Anonymity, Flx, Protocol};
13//!
14//! # async fn example() -> anyhow::Result<()> {
15//! let proxies = Flx::fetch()
16//!     .types([Protocol::Http(Anonymity::Elite)])
17//!     .limit(20)
18//!     .collect()
19//!     .await?;
20//! # Ok(())
21//! # }
22//! ```
23//!
24//! See [`Flx`] for the full builder contract, `examples/` for runnable
25//! workflows, and the README for CLI usage.
26//!
27//! ## Features
28//!
29//! - `log` (default): emit `flx::*` records via the `log` crate.
30//! - `progress_bar` (default): CLI progress rendering.
31//! - `clap` (default): enables the `flx` binary.
32//! - `serve` (optional): rotating proxy endpoint (`flx serve`);
33//!   off by default while experimental.
34//!
35//! ## Minimum Supported Rust Version
36//!
37//! Requires a recent stable toolchain (tested on 1.89+). No `rust-version`
38//! floor is declared in `Cargo.toml`; MSRV bumps are treated as a minor
39//! breaking change and noted in release notes.
40//!

42//!

116/// Sends log records to `path` instead of stderr, for runs that own the screen.
117///
118/// ANSI is always off: the destination is a file to be paged through later, and
119/// anything written to stderr while a full-screen UI is up would corrupt it.
120#[cfg(feature = "log")]
121pub fn initialize_file_logging(
122    log_level: log::LevelFilter,
123    path: &std::path::Path,
124) -> anyhow::Result<()> {
125    use anyhow::Context as _;
126    let file = std::fs::OpenOptions::new()
127        .create(true)
128        .append(true)
129        .open(path)
130        .with_context(|| format!("cannot open the log file {}", path.display()))?;
131    log::set_boxed_logger(Box::new(FlxLogger::to_file(file)))?;
132    log::set_max_level(log_level);
133    Ok(())
134}

147/// Where flx records go: the terminal, or a file for a full-screen run.
148#[cfg(feature = "log")]
149enum LogSink {
150    Stderr,
151    /// Appended to, behind a lock: many tasks log from many threads.
152    File(std::sync::Mutex<std::fs::File>),
153}
154
155/// Writes flx records with tty-gated colors.
156#[cfg(feature = "log")]
157struct FlxLogger {
158    sink: LogSink,
159}
160
161#[cfg(feature = "log")]
162const LOG_MODULE_ROOT: &str = "flx";
163
164#[cfg(feature = "log")]
165fn log_module_allowed(target: &str) -> bool {
166    match target.strip_prefix(LOG_MODULE_ROOT) {
167        // Matches Rust module paths split by `::`.
168        Some(rest) => rest.is_empty() || rest.starts_with("::"),
169        None => false,
170    }
171}

173#[cfg(feature = "log")]
174fn log_prefix_color(level: log::Level) -> &'static str {
175    match level {
176        log::Level::Error => "\x1b[31m",
177        log::Level::Warn => "\x1b[33m",
178        log::Level::Info => "\x1b[34m",
179        log::Level::Debug => "\x1b[36m",
180        log::Level::Trace => "\x1b[35m",
181    }
182}

184#[cfg(feature = "log")]
185impl FlxLogger {
186    fn to_stderr() -> Self {
187        Self {
188            sink: LogSink::Stderr,
189        }
190    }
191
192    fn to_file(file: std::fs::File) -> Self {
193        Self {
194            sink: LogSink::File(std::sync::Mutex::new(file)),
195        }
196    }
197
198    fn color_enabled() -> bool {
199        use std::io::IsTerminal as _;
200        match std::env::var_os("TERM") {
201            None => return false,
202            Some(term) => {
203                if term == "dumb" {
204                    return false;
205                }
206            }
207        }
208        if std::env::var_os("NO_COLOR").is_some() {
209            return false;
210        }
211        std::io::stderr().is_terminal()
212    }
213
214    /// One record, as a single line: the color is only added for a terminal.
215    fn write_record(&self, record: &log::Record, color: bool) -> std::io::Result<()> {
216        match &self.sink {
217            LogSink::Stderr => {
218                let mut stderr = std::io::stderr().lock();
219                write_record_to(&mut stderr, record, color)
220            }
221            LogSink::File(file) => {
222                let mut file = file.lock().unwrap_or_else(|e| e.into_inner());
223                write_record_to(&mut *file, record, color)
224            }
225        }
226    }
227}

229#[cfg(feature = "log")]
230fn write_record_to(
231    target: &mut impl std::io::Write,
232    record: &log::Record,
233    color: bool,
234) -> std::io::Result<()> {
235    if color {
236        // Keeps prefix bytes identical to the previous logger.
237        write!(
238            target,
239            "\x1b[0m{}{}: {} ",
240            log_prefix_color(record.level()),
241            record.target(),
242            record.level()
243        )?;
244        write!(target, "\x1b[0m")?;
245    } else {
246        write!(target, "{}: {} ", record.target(), record.level())?;
247    }
248    writeln!(target, "{}", record.args())
249}

251#[cfg(feature = "log")]
252impl log::Log for FlxLogger {
253    fn enabled(&self, metadata: &log::Metadata) -> bool {
254        metadata.level() <= log::max_level() && log_module_allowed(metadata.target())
255    }
256
257    fn log(&self, record: &log::Record) {
258        if !self.enabled(record.metadata()) {
259            return;
260        }
261        let color = matches!(self.sink, LogSink::Stderr) && Self::color_enabled();
262        let _ = self.write_record(record, color);
263    }
264
265    fn flush(&self) {
266        match &self.sink {
267            LogSink::Stderr => {
268                let _ = std::io::stderr().flush();
269            }
270            LogSink::File(file) => {
271                if let Ok(mut file) = file.lock() {
272                    let _ = file.flush();
273                }
274            }
275        }
276    }
277}

279/// Reads proxies from files with per-line protocol pinning.
280///
281/// Bare `ip:port` lines inherit all requested protocols; scheme-prefixed
282/// lines (`socks5://…`) pin their own type. `-` means stdin.
283/// See [`load_proxy_files`] for the file-loading helper.
284///
285/// # Examples
286///
287/// ```no_run
288/// use flx::ProxySource;
289///
290/// # fn example() -> anyhow::Result<()> {
291/// let source = ProxySource::from_reader(std::io::Cursor::new("1.2.3.4:8080\n"))?;
292/// let proxies: Vec<_> = source.collect();
293/// # Ok(())
294/// # }
295/// ```
296pub struct ProxySource {
297    lines: Lines<Box<dyn BufRead + Send>>,
298    default_proxy_types: Arc<[Protocol]>,
299}
300
301/// Defines fallback protocols for bare ip:port lines.
302static FILE_DEFAULT_PROTOCOLS: LazyLock<Arc<[Protocol]>> = LazyLock::new(|| {
303    Arc::from([
304        Protocol::Http(Anonymity::Unknown),
305        Protocol::Https(Anonymity::Unknown),
306        Protocol::Socks4,
307        Protocol::Socks5,
308    ])
309});
310
311/// Formats args into buf, borrowing when it fits.
312pub(crate) fn write_to_buffer<'a>(
313    buf: &'a mut [u8],
314    args: std::fmt::Arguments<'_>,
315) -> Cow<'a, str> {
316    let mut writer = Cursor::new(buf);
317    match writer.write_fmt(args) {
318        Ok(()) => {
319            let len = writer.position() as usize;
320            Cow::Borrowed(std::str::from_utf8(&writer.into_inner()[..len]).expect("ASCII"))
321        }
322        Err(_) => Cow::Owned(args.to_string()),
323    }
324}

326impl ProxySource {
327    /// Builds a fetcher-backed source from `config` without blocking.
328    ///
329    /// # Errors
330    ///
331    /// Returns an error when the provider set cannot be assembled.
332    pub async fn from_fetcher(config: FetcherConfig) -> anyhow::Result<ProxyFetcher> {
333        ProxyFetcher::gather(config).await
334    }
335
336    /// Opens `filepath` for lazy line-by-line parsing.
337    ///
338    /// # Errors
339    ///
340    /// Returns an I/O error when the file cannot be opened.
341    pub fn from_file(filepath: PathBuf) -> anyhow::Result<Self> {
342        let file = anyhow::Context::with_context(File::open(&filepath), || {
343            format!("failed to open proxy file {}", filepath.display())
344        })?;
345        Self::from_reader(BufReader::new(file))
346    }
347
348    /// Wraps any buffered reader as a proxy source.
349    ///
350    /// # Errors
351    ///
352    /// Currently infallible; returns `Ok` for API symmetry with [`ProxySource::from_file`].
353    pub fn from_reader<R: BufRead + Send + 'static>(reader: R) -> anyhow::Result<Self> {
354        let lines = (Box::new(reader) as Box<dyn BufRead + Send>).lines();
355
356        let default_proxy_types = Arc::clone(&FILE_DEFAULT_PROTOCOLS);
357
358        Ok(Self {
359            lines,
360            default_proxy_types,
361        })
362    }
363
364    /// Reads candidates from stdin without blocking.
365    ///
366    /// # Errors
367    ///
368    /// Currently infallible; returns `Ok` for API symmetry with [`ProxySource::from_file`].
369    pub fn from_stdin() -> anyhow::Result<Self> {
370        Self::from_reader(std::io::BufReader::new(std::io::stdin()))
371    }
372}

374impl Iterator for ProxySource {
375    type Item = Proxy;
376
377    fn next(&mut self) -> Option<Self::Item> {
378        for line in self.lines.by_ref() {
379            let line = match line {
380                Ok(line) => line,
381                Err(error) => {
382                    #[cfg(feature = "log")]
383                    log::warn!("failed to read a line from the proxy file: {error}");
384                    #[cfg(not(feature = "log"))]
385                    let _ = error;
386                    continue;
387                }
388            };
389            let line = line.trim();
390            if line.is_empty() {
391                continue;
392            }
393            let mut proxy = match line.parse::<Proxy>() {
394                Ok(proxy) => proxy,
395                Err(error) => {
396                    #[cfg(feature = "log")]
397                    log::warn!("skipped unparseable proxy file line '{line}': {error}");
398                    #[cfg(not(feature = "log"))]
399                    let _ = error;
400                    continue;
401                }
402            };
403            // Pins scheme-prefixed lines; bare lines inherit defaults.
404            if proxy.expected_types.is_empty() {
405                proxy.expected_types = Arc::clone(&self.default_proxy_types);
406            }
407            return Some(proxy);
408        }
409        None
410    }
411}
