14/// Negotiates SOCKS4 proxy connections.
15pub struct Socks4Negotiator;
16
17impl Socks4Negotiator {
18    fn build_connect_request<'a>(buf: &'a mut [u8], host: &str, port: u16) -> Cow<'a, [u8]> {
19        // 2 (VN, CD) + 2 (port) + 4 (DST.IP) + 1 (USERID null) + host + 1.
20        let total = 10 + host.len();
21        if total > buf.len() {
22            // Fallback for pathologically overlong hosts (domain format).
23            let mut packet = Vec::with_capacity(total);
24            packet.extend_from_slice(&[4u8, 1u8]); // VN, command
25            packet.extend_from_slice(&port.to_be_bytes());
26            packet.extend_from_slice(&[0, 0, 0, 1]);
27            packet.push(0u8);
28            packet.extend_from_slice(host.as_bytes());
29            packet.push(0u8);
30            return Cow::Owned(packet);
31        }
32
33        let mut len = 0usize;
34        buf[len..len + 2].copy_from_slice(&[4u8, 1u8]); // VN, command
35        len += 2;
36        buf[len..len + 2].copy_from_slice(&port.to_be_bytes());
37        len += 2;
38        match host.parse::<std::net::Ipv4Addr>() {
39            Ok(ip) => {
40                buf[len..len + 4].copy_from_slice(&ip.octets());
41                len += 4;
42                buf[len] = 0u8;
43                len += 1;
44            }
45            Err(_) => {
46                buf[len..len + 4].copy_from_slice(&[0, 0, 0, 1]);
47                len += 4;
48                buf[len] = 0u8;
49                len += 1;
50                buf[len..len + host.len()].copy_from_slice(host.as_bytes());
51                len += host.len();
52                buf[len] = 0u8;
53                len += 1;
54            }
55        }
56        Cow::Borrowed(&buf[..len])
57    }
58}

60#[async_trait]
61impl NegotiatorTrait for Socks4Negotiator {
62    async fn negotiate(
63        &self,
64        stream: &mut TcpStream,
65        _proxy_host: &str,
66        uri: &Uri,
67    ) -> anyhow::Result<()> {
68        let host = uri.host().context("SOCKS4 target URI has no host")?;
69        let port = uri
70            .port_u16()
71            .or_else(|| match uri.scheme_str() {
72                Some("http") => Some(80),
73                Some("https") => Some(443),
74                _ => None,
75            })
76            .context("SOCKS4 target URI has no port")?;
77
78        // Build SOCKS4 CONNECT request; use domain fallback for non-IPv4 hosts.
79        let mut packet_buf = [0u8; 512];
80        let packet = Self::build_connect_request(&mut packet_buf, host, port);
81
82        stream.write_all(&packet).await?;
83
84        let mut response = [0u8; 8];
85        stream.read_exact(&mut response).await?;
86
87        // Reply layout is [VN, CD, DST.PORT, DST.IP]; CD signals success.
88        let mut response_slice = &response[..];
89        if response_slice.read_u8().await? != 0 {
90            anyhow::bail!("InvalidData: invalid response version");
91        }
92
93        match response_slice.read_u8().await? {
94            90 => {}
95            91 => anyhow::bail!("Other: Request rejected or failed"),
96            92 => anyhow::bail!("PermissionDenied: Request rejected because SOCKS server cannot connect to identd on the client"),
97            93 => anyhow::bail!("PermissionDenied: Request rejected because the client program and identd report different user IDs"),
98            code => anyhow::bail!("InvalidData: invalid response code: {}", code),
99        }
100
101        Ok(())
102    }
103}
