17const SCHEMA_VERSION: u8 = 1;
18const DEFAULT_FILE_NAME: &str = "health.jsonl";
19const MAX_RECORD_BYTES: usize = 16 * 1024;

21/// Persistent health history for proxy endpoints.
22///
23/// The file handle is reopened on demand: a write racing runtime shutdown
24/// poisons the `tokio::fs::File` it hit, so the poisoned handle is dropped
25/// and replaced instead of reused.
26#[derive(Clone)]
27pub struct HealthStore {
28    path: Arc<PathBuf>,
29    state: Arc<Mutex<HashMap<Endpoint, HealthStats>>>,
30    writer: Arc<tokio::sync::Mutex<tokio::fs::File>>,
31}
32
33/// Aggregated health statistics for one endpoint.
34#[derive(Clone, Debug, Default, PartialEq)]
35pub struct HealthStats {
36    /// Number of successful probes.
37    pub successes: u64,
38    /// Number of failed probes.
39    pub failures: u64,
40    /// Total successful probe latency in seconds.
41    pub latency_total: f64,
42    /// Best HTTP(S) anonymity observed.
43    pub anonymity: Option<Anonymity>,
44    /// Unix timestamp of the latest recorded event.
45    pub last_checked: f64,
46}
47
48/// A score calculated from persistent endpoint health.
49#[derive(Clone, Copy, Debug, PartialEq)]
50pub struct HealthScore {
51    /// Reliability component in the range 0–100.
52    pub reliability: f64,
53    /// Speed component in the range 0–100.
54    pub speed: f64,
55    /// Anonymity component in the range 0–100.
56    pub anonymity: f64,
57    /// Weighted final score in the range 0–100.
58    pub total: f64,
59}

83#[derive(Debug, Deserialize)]
84struct StoredRecord {
85    schema: u8,
86    timestamp: f64,
87    ip: Ipv4Addr,
88    port: u16,
89    protocol: String,
90    success: bool,

95impl HealthStore {
96    /// Opens the default persistent health store.
97    pub async fn open_default() -> Result<Self> {
98        let path = crate::geolookup::data_dir()?
99            .join("health")
100            .join(DEFAULT_FILE_NAME);
101        Self::open(path).await
102    }
103
104    /// Opens a persistent health store at `path`.
105    pub async fn open(path: impl Into<PathBuf>) -> Result<Self> {
106        let path = path.into();
107        if let Some(parent) = path.parent() {
108            tokio::fs::create_dir_all(parent).await.with_context(|| {
109                format!("failed to create health directory {}", parent.display())
110            })?;
111        }
112        let state = load_state(&path).await?;
113        let writer = tokio::fs::OpenOptions::new()
114            .create(true)
115            .append(true)
116            .open(&path)
117            .await
118            .with_context(|| format!("failed to open health store {}", path.display()))?;
119        Ok(Self {
120            path: Arc::new(path),
121            state: Arc::new(Mutex::new(state)),
122            writer: Arc::new(tokio::sync::Mutex::new(writer)),
123        })
124    }
125
126    /// Returns the backing file path.
127    pub fn path(&self) -> &Path {
128        &self.path
129    }
130
131    /// Records one successful probe.
132    pub async fn record_success(
133        &self,
134        ip: Ipv4Addr,
135        port: u16,
136        protocol: Protocol,
137        latency_secs: f64,
138        anonymity: Option<Anonymity>,
139    ) -> Result<()> {
140        self.record(HealthRecord {
141            schema: SCHEMA_VERSION,
142            timestamp: now(),
143            ip,
144            port,
145            protocol: protocol.to_string(),
146            success: true,
147            latency_secs: Some(latency_secs),
148            anonymity,
149            reason: None,
150        })
151        .await
152    }
153
154    /// Records one failed probe.
155    pub async fn record_failure(
156        &self,
157        ip: Ipv4Addr,
158        port: u16,
159        protocol: Protocol,
160        reason: &str,
161    ) -> Result<()> {
162        self.record(HealthRecord {
163            schema: SCHEMA_VERSION,
164            timestamp: now(),
165            ip,
166            port,
167            protocol: protocol.to_string(),
168            success: false,
169            latency_secs: None,
170            anonymity: None,
171            reason: Some(reason.to_owned()),
172        })
173        .await
174    }
175
176    /// Returns aggregated statistics for an endpoint.
177    pub fn stats(&self, ip: Ipv4Addr, port: u16) -> Option<HealthStats> {
178        self.state
179            .lock()
180            .unwrap_or_else(|error| error.into_inner())
181            .get(&Endpoint { ip, port })
182            .cloned()
183    }
184
185    /// Calculates the score for an endpoint, if it has history.
186    pub fn score(&self, ip: Ipv4Addr, port: u16) -> Option<HealthScore> {
187        self.stats(ip, port).map(score_stats)
188    }
189
190    async fn record(&self, record: HealthRecord) -> Result<()> {
191        let mut body = serde_json::to_vec(&record).context("failed to serialize health record")?;
192        if body.len() > MAX_RECORD_BYTES {
193            anyhow::bail!("health record exceeds {MAX_RECORD_BYTES} bytes");
194        }
195        body.push(b'\n');
196        if let Err(error) = self.append(&body).await {
197            if let Some(replacement) =
198                crate::file_util::reopen_after_shutdown(&self.path, &error).await?
199            {
200                *self.writer.lock().await = replacement;
201                self.append(&body).await?;
202            } else {
203                return Err(error);
204            }
205        }
206        let mut state = self.state.lock().unwrap_or_else(|error| error.into_inner());
207        apply_record(&mut state, &record);
208        Ok(())
209    }
210
211    async fn append(&self, body: &[u8]) -> Result<()> {
212        let mut writer = self.writer.lock().await;
213        writer
214            .write_all(body)
215            .await
216            .context("failed to append health record")?;
217        writer
218            .flush()
219            .await
220            .context("failed to flush health record")?;
221        Ok(())
222    }
223}

225fn score_stats(stats: HealthStats) -> HealthScore {
226    let total = stats.successes.saturating_add(stats.failures);
227    let reliability = if total == 0 {
228        0.0
229    } else {
230        stats.successes as f64 / total as f64 * 100.0
231    };
232    let avg_latency = if stats.successes == 0 {
233        5.0
234    } else {
235        stats.latency_total / stats.successes as f64
236    };
237    let speed = (100.0 * (1.0 - avg_latency / 5.0)).clamp(0.0, 100.0);
238    let anonymity = stats
239        .anonymity
240        .map(|level| match level {
241            Anonymity::Transparent => 0.0,
242            Anonymity::Anonymous => 50.0,
243            Anonymity::Elite => 100.0,
244            Anonymity::Unknown => 0.0,
245        })
246        .unwrap_or(0.0);
247    HealthScore {
248        reliability,
249        speed,
250        anonymity,
251        total: (reliability * 0.60 + speed * 0.25 + anonymity * 0.15).clamp(0.0, 100.0),
252    }
253}
254
255fn apply_record(state: &mut HashMap<Endpoint, HealthStats>, record: &HealthRecord) {
256    let entry = state
257        .entry(Endpoint {
258            ip: record.ip,
259            port: record.port,
260        })
261        .or_default();
262    if record.success {
263        entry.successes = entry.successes.saturating_add(1);
264        if let Some(latency) = record
265            .latency_secs
266            .filter(|value| value.is_finite() && *value >= 0.0)
267        {
268            entry.latency_total += latency;
269        }
270        if let Some(anonymity) = record.anonymity {
271            if entry
272                .anonymity
273                .is_none_or(|current| anonymity.rank() > current.rank())
274            {
275                entry.anonymity = Some(anonymity);
276            }
277        }
278    } else {
279        entry.failures = entry.failures.saturating_add(1);
280    }
281    entry.last_checked = record.timestamp;
282}

284async fn load_state(path: &Path) -> Result<HashMap<Endpoint, HealthStats>> {
285    let body = match tokio::fs::read_to_string(path).await {
286        Ok(body) => body,
287        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(HashMap::new()),
288        Err(error) => {
289            return Err(error)
290                .with_context(|| format!("failed to read health store {}", path.display()))
291        }
292    };
293    let mut state = HashMap::new();
294    for line in body.lines() {
295        if line.trim().is_empty() {
296            continue;
297        }
298        match serde_json::from_str::<StoredRecord>(line) {
299            Ok(record) if record.schema == SCHEMA_VERSION => {
300                let record = HealthRecord {
301                    schema: record.schema,
302                    timestamp: record.timestamp,
303                    ip: record.ip,
304                    port: record.port,
305                    protocol: record.protocol,
306                    success: record.success,
307                    latency_secs: record.latency_secs,
308                    anonymity: record.anonymity,
309                    reason: None,
310                };
311                apply_record(&mut state, &record);
312            }
313            Ok(_) => warn_corrupt_record("unknown schema"),
314            Err(_) => warn_corrupt_record("invalid JSON"),
315        }
316    }
317    Ok(state)
318}
319
320fn warn_corrupt_record(reason: &str) {
321    #[cfg(feature = "log")]
322    log::warn!("ignored health record: {reason}");
323    #[cfg(not(feature = "log"))]
324    let _ = reason;
325}
326
327fn now() -> f64 {
328    SystemTime::now()
329        .duration_since(UNIX_EPOCH)
330        .map(|duration| duration.as_secs_f64())
331        .unwrap_or(0.0)
332}

334#[cfg(test)]

337    use std::sync::atomic::{AtomicU64, Ordering};

339    fn path(stem: &str) -> PathBuf {
340        static NEXT: AtomicU64 = AtomicU64::new(0);
341        std::env::temp_dir().join(format!(
342            "flx_health_{stem}_{}_{}.jsonl",
343            std::process::id(),
344            NEXT.fetch_add(1, Ordering::Relaxed)
345        ))
346    }

348    #[tokio::test]
349    async fn empty_store_has_no_score() {
350        let store = HealthStore::open(path("empty")).await.unwrap();
351        assert!(store.score(Ipv4Addr::LOCALHOST, 8080).is_none());
352    }

354    #[tokio::test]
355    async fn repeated_events_reload_and_calculate_score() {
356        let file = path("reload");
357        let store = HealthStore::open(&file).await.unwrap();
358        store
359            .record_success(
360                Ipv4Addr::LOCALHOST,
361                8080,
362                Protocol::Http(Anonymity::Elite),
363                0.5,
364                Some(Anonymity::Elite),
365            )
366            .await
367            .unwrap();
368        store
369            .record_failure(Ipv4Addr::LOCALHOST, 8080, Protocol::Socks5, "timeout")
370            .await
371            .unwrap();
372        let score = store.score(Ipv4Addr::LOCALHOST, 8080).unwrap();
373        assert_eq!(score.reliability, 50.0);
374        assert_eq!(score.speed, 90.0);
375        assert_eq!(score.anonymity, 100.0);
376        assert_eq!(score.total, 67.5);
377
378        let reloaded = HealthStore::open(file).await.unwrap();
379        assert_eq!(
380            reloaded.stats(Ipv4Addr::LOCALHOST, 8080).unwrap().failures,
381            1
382        );
383    }
