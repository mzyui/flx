1use std::time::Duration;
2
3use async_trait::async_trait;
4
5use super::models::{valid_sources, ScrapeMode, Source};
6use super::ProxyProvider;
7
8/// Scrapes proxy-list.org paged listings.
9pub struct ProxyListOrgProvider;
10
11const MAX_PAGES: u32 = 10;
12
13#[async_trait]
14impl ProxyProvider for ProxyListOrgProvider {
15    fn name(&self) -> &'static str {
16        "proxylist-org"
17    }
18
19    fn sources(&self) -> Vec<Source> {
20        valid_sources(
21            (1..=MAX_PAGES)
22                .map(|page| {
23                    let url = format!("https://proxy-list.org/english/index.php?p={}", page);
24                    Source::http(&url).map(|source| {
25                        source
26                            .with_mode(ScrapeMode::Base64Rows)
27                            .with_timeout(Duration::from_secs(15))
28                    })
29                })
30                .collect(),
31        )
32    }
33}
