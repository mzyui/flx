12const MAX_JUDGE_RESPONSE_BYTES: usize = 576 * 1024;
13
14pub(super) async fn verify_tls_judge(
15    stream: TcpStream,
16    target: &super::JudgeTarget,
17    authority: &str,
18    params: &WorkParams,
19) -> anyhow::Result<(String, Option<ConnectionDriver>)> {
20    let insecure = params.insecure;
21    let support_cookies = params.support_cookies;
22    let support_referer = params.support_referer;
23    use hyper::client::conn::http1::handshake;
24    use hyper_util::rt::TokioIo;
25
26    use crate::proxy::client::tls_connect;
27    let tls_stream = tls_connect(&target.host, stream, insecure)
28        .await
29        .with_context(|| format!("TLS handshake with judge {} failed", target.host))?;
30    let (mut sender, connection) = handshake(TokioIo::new(tls_stream))
31        .await
32        .context("HTTP handshake with TLS judge failed")?;
33    let driver = spawn_connection_driver(connection, Arc::from(authority), Duration::from_secs(30));
34    let request = hyper::Request::get(&target.path_and_query)
35        .header(hyper::header::HOST, authority)
36        .header(hyper::header::CONNECTION, "close")
37        .header("X-Fluxy-Token", &target.request_token)
38        .header(hyper::header::COOKIE, "cookie=ok")
39        .header(hyper::header::REFERER, "https://google.com/")
40        .body(http_body_util::Empty::<hyper::body::Bytes>::new())?;
41    let response = sender.send_request(request).await?;
42    if response.status() != hyper::StatusCode::OK {
43        anyhow::bail!("TLS judge returned status {}", response.status());
44    }
45    let body = read_bounded_body(response.into_body(), MAX_JUDGE_RESPONSE_BYTES).await?;
46    if body.len() > MAX_JUDGE_RESPONSE_BYTES {
47        anyhow::bail!("TLS judge response exceeds validation limit");
48    }
49    // Scan ASCII token with single substring search.
50    let body = String::from_utf8_lossy(&body);
51    if !body.contains(&target.response_marker) {
52        anyhow::bail!("response did not originate from the TLS judge");
53    }
54    if support_cookies && !body.contains("HTTP_COOKIE = cookie=ok") {
55        anyhow::bail!("proxy did not forward the cookie header");
56    }
57    if support_referer && !body.contains("HTTP_REFERER = https://google.com/") {
58        anyhow::bail!("proxy did not forward the referer header");
59    }
60    Ok((body.into_owned(), Some(driver)))
61}

63pub(super) async fn verify_judge(
64    stream: &mut BufReader<TcpStream>,
65    target: &super::JudgeTarget,
66    authority: &str,
67    params: &WorkParams,
68) -> anyhow::Result<String> {
69    let support_cookies = params.support_cookies;
70    let support_referer = params.support_referer;
71    let mut buf = [0u8; 2048];
72    let request = crate::write_to_buffer(
73        &mut buf,
74        format_args!(
75            "GET {} HTTP/1.1\r\nHost: {}\r\nX-Fluxy-Token: {}\r\nCookie: cookie=ok\r\nReferer: https://google.com/\r\nConnection: close\r\n\r\n",
76            target.path_and_query, authority, target.request_token
77        ),
78    );
79    stream.write_all(request.as_bytes()).await?;
80
81    let mut response = Vec::with_capacity(1024);
82    let mut chunk = [0u8; 4096];
83    loop {
84        let read = stream.read(&mut chunk).await?;
85        if read == 0 {
86            break;
87        }
88        if response.len().saturating_add(read) > MAX_JUDGE_RESPONSE_BYTES {
89            anyhow::bail!("local judge response exceeds validation limit");
90        }
91        response.extend_from_slice(&chunk[..read]);
92    }
93
94    let status = super::negotiate::parse_http_status(&response)?;
95    if status != 200 {
96        anyhow::bail!("local judge returned status {status} through tunnel");
97    }
98    let response = String::from_utf8_lossy(&response);
99    if !response.contains(&target.response_marker) {
100        anyhow::bail!("response did not originate from the local judge");
101    }
102    if support_cookies && !response.contains("HTTP_COOKIE = cookie=ok") {
103        anyhow::bail!("proxy did not forward the cookie header");
104    }
105    if support_referer && !response.contains("HTTP_REFERER = https://google.com/") {
106        anyhow::bail!("proxy did not forward the referer header");
107    }
108    Ok(response.into_owned())
109}
