67fn protocol_matches<F>(a: &Protocol, b: &Protocol, on_http_https: F) -> bool
68where
69    F: FnOnce(&Anonymity, &Anonymity) -> bool,
70{
71    match (a, b) {
72        (Protocol::Http(left), Protocol::Http(right))
73        | (Protocol::Https(left), Protocol::Https(right)) => on_http_https(left, right),
74        (Protocol::Connect(left), Protocol::Connect(right)) => left == right,
75        _ => a == b,
76    }
77}

79pub(crate) fn advertised_matches_request(advertised: &Protocol, requested: &Protocol) -> bool {
80    protocol_matches(advertised, requested, |left, right| {
81        matches!(left, Anonymity::Unknown) || matches!(right, Anonymity::Unknown) || left == right
82    })
83}

85/// Identity of an actual probe.
86///
87/// HTTP(S) anonymity is *measured* by the probe, never used to perform it, so
88/// it is erased here; Socks/Connect variants keep their exact shape. Two jobs
89/// with the same probe family and requested level probe identically.
90#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
91pub(crate) enum ProbeFamily {
92    Http,
93    Https,
94    Socks4,
95    Socks5,
96    Connect(u16),
97}
98
99pub(crate) fn probe_family(protocol: &Protocol) -> ProbeFamily {
100    match protocol {
101        Protocol::Http(_) => ProbeFamily::Http,
102        Protocol::Https(_) => ProbeFamily::Https,
103        Protocol::Socks4 => ProbeFamily::Socks4,
104        Protocol::Socks5 => ProbeFamily::Socks5,
105        Protocol::Connect(port) => ProbeFamily::Connect(*port),
106    }
107}

109/// Builds the unique `(probe protocol, requested protocol)` pairs for one proxy.
110///
111/// HTTP(S) anonymity is measured by the probe, so advertised levels of the same
112/// family collapse into a single probe; the requested level stays exact. With
113/// `probe_missed`, only requested types the advertisement does not cover are
114/// probed (every request when nothing is advertised).
115pub(crate) fn singleton_jobs(
116    advertised: &[Protocol],
117    requested: &[Protocol],
118    probe_missed: bool,
119) -> Vec<(Protocol, Protocol)> {
120    if probe_missed {
121        let mut seen: Vec<Protocol> = Vec::with_capacity(requested.len());
122        let mut jobs = Vec::new();
123        for requested in requested {
124            if seen.contains(requested) {
125                continue;
126            }
127            seen.push(*requested);
128            let covered = advertised
129                .iter()
130                .any(|adv| advertised_matches_request(adv, requested));
131            if !covered {
132                jobs.push((*requested, *requested));
133            }
134        }
135        jobs
136    } else {
137        let mut seen_keys: HashSet<(ProbeFamily, Protocol)> = HashSet::new();
138        let mut seen_adv: Vec<Protocol> = Vec::with_capacity(advertised.len());
139        let mut jobs = Vec::new();
140        for adv in advertised {
141            if seen_adv.contains(adv) {
142                continue;
143            }
144            seen_adv.push(*adv);
145            let mut seen_req: Vec<Protocol> = Vec::with_capacity(requested.len());
146            for req in requested {
147                if seen_req.contains(req) {
148                    continue;
149                }
150                seen_req.push(*req);
151                if !advertised_matches_request(adv, req) {
152                    continue;
153                }
154                if seen_keys.insert((probe_family(adv), *req)) {
155                    jobs.push((*adv, *req));
156                }
157            }
158        }
159        jobs
160    }
161}

163pub(crate) fn result_satisfies_request(result: &Protocol, requested: &Protocol) -> bool {
164    protocol_matches(result, requested, |actual, required| {
165        matches!(required, Anonymity::Unknown) || actual == required
166    })
167}

169async fn run_probe(
170    proxy: &mut Proxy,
171    protocol: Protocol,
172    requested: Protocol,
173    targets: &JudgeTargets,
174    params: &WorkParams,
175) -> anyhow::Result<Option<ProxyType>> {
176    if let Protocol::Http(_) = protocol {
177        let result = checker::support_http(proxy, &targets.http, params)
178            .await
179            .with_context(|| format!("{}: HTTP check failed", proxy.as_text()))?;
180        if let Some(result) =
181            result.filter(|result| result_satisfies_request(&result.inner, &requested))
182        {
183            result.apply(proxy);
184            Ok(Some(ProxyType::checked(result.inner)))
185        } else {
186            Ok(None)
187        }
188    } else {
189        let result = tunnel::support_tunnel(proxy, protocol, &targets.tunnel, params)
190            .await
191            .with_context(|| format!("{}: tunnel check failed", proxy.as_text()))?;
192        if let Some(result) =
193            result.filter(|result| result_satisfies_request(&result.inner, &requested))
194        {
195            result.apply(proxy);
196            Ok(Some(ProxyType::checked(result.inner)))
197        } else {
198            Ok(None)
199        }
200    }
201}

203fn classify_failure(error: &anyhow::Error, protocol: Protocol) -> String {
204    let text = format!("{error:#}").to_ascii_lowercase();
205    if text.contains("timed out") {
206        "timeout".to_owned()
207    } else if text.contains("did not originate")
208        || text.contains("returned status")
209        || text.contains("rejected request")
210        || text.contains("did not accept")
211        || text.contains("did not forward")
212    {
213        "rejected".to_owned()
214    } else {
215        format!("error ({protocol}): {text}")
216    }
217}

219fn report_failure(
220    sender: &Option<mpsc::Sender<ProxyFailure>>,
221    proxy: &Proxy,
222    protocol: Protocol,
223    reason: String,
224) {
225    if let Some(sender) = sender {
226        let _ = sender.try_send(ProxyFailure {
227            ip: proxy.ip,
228            port: proxy.port,
229            protocol,
230            reason,
231        });
232    }

235pub(crate) async fn do_work(
236    job: SingletonJob,
237    sender: mpsc::Sender<Proxy>,
238    counters: ValidationProgress,
239    targets: JudgeTargets,
240    params: &WorkParams,
241    failures: Option<mpsc::Sender<ProxyFailure>>,
242) -> anyhow::Result<()> {
243    let SingletonJob {
244        proxy,
245        protocol,
246        requested,
247    } = job;
248    let mut proxy = proxy.validation_probe();
249    let result = run_probe(&mut proxy, protocol, requested, &targets, params).await;
250    counters
251        .done
252        .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
253    match result {
254        Ok(Some(proxy_type)) => {
255            proxy.proxy_types.push(proxy_type);
256            counters
257                .passed
258                .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
259            let _ = sender.send(proxy).await;
260        }
261        Ok(None) => report_failure(&failures, &proxy, protocol, "unsatisfied".to_owned()),
262        Err(error) => report_failure(
263            &failures,
264            &proxy,
265            protocol,
266            classify_failure(&error, protocol),
267        ),
268    }
269    Ok(())
270}

283pub(crate) async fn do_group_work(
284    member: GroupMemberJob,
285    group_tx: mpsc::Sender<GroupWorkResult>,
286    targets: JudgeTargets,
287    params: &WorkParams,
288    failures: Option<mpsc::Sender<ProxyFailure>>,
289    dead_map: Option<GroupDeadMap>,
290) -> anyhow::Result<()> {
291    let GroupMemberJob {
292        proxy,
293        proxy_id,
294        protocol,
295        group_idx,
296        slot,
297        group_len,
298    } = member;
299    let key = GroupKey {
300        proxy_id,
301        group_idx,
302    };
303    if let Some(map) = dead_map.as_ref() {
304        let dead = {
305            let mut guard = map.lock().unwrap_or_else(|e| e.into_inner());
306            guard
307                .entry(key)
308                .or_insert_with(|| Arc::new(AtomicBool::new(false)))
309                .clone()
310        };
311        if dead.load(Ordering::Relaxed) {
312            let probe = proxy.validation_probe();
313            report_failure(&failures, &probe, protocol, "group-dead".to_owned());
314            let _ = group_tx
315                .send(GroupWorkResult {
316                    key,
317                    slot,
318                    group_len,
319                    proxy: None,
320                })
321                .await;
322            return Ok(());
323        }
324    }
325    let mut probe = proxy.validation_probe();
326    let result = match run_probe(&mut probe, protocol, protocol, &targets, params).await {
327        Ok(Some(proxy_type)) => {
328            probe.proxy_types.push(proxy_type);
329            Some(probe)
330        }
331        Ok(None) => {
332            report_failure(&failures, &probe, protocol, "unsatisfied".to_owned());
333            None
334        }
335        Err(error) => {
336            report_failure(
337                &failures,
338                &probe,
339                protocol,
340                classify_failure(&error, protocol),
341            );
342            None
343        }
344    };
345    if result.is_none() {
346        if let Some(map) = dead_map.as_ref() {
347            if let Some(dead) = map.lock().unwrap_or_else(|e| e.into_inner()).get(&key) {
348                dead.store(true, Ordering::Relaxed);
349            }
350        }
351    }
352    let _ = group_tx
353        .send(GroupWorkResult {
354            key,
355            slot,
356            group_len,
357            proxy: result,
358        })
359        .await;
360    Ok(())
361}

363pub(crate) fn group_finish(state: GroupState) -> Option<Proxy> {
364    let slots: Vec<Proxy> = if state.results.iter().all(Option::is_some) {
365        state.results.into_iter().flatten().collect()
366    } else {
367        return None;
368    };
369    let mut merged = slots[0].clone();
370    merged.proxy_types = slots
371        .iter()
372        .filter_map(|proxy| proxy.proxy_types.first().cloned())
373        .collect();
374    merged.runtimes = RuntimeStats::default();
375    for slot in &slots {
376        let avg = slot.runtimes.avg();
377        if avg > 0.0 {
378            merged.runtimes.record(avg);
379        }
380    }
381    Some(merged)
382}

384pub(crate) async fn aggregate_groups(
385    mut group_rx: mpsc::Receiver<GroupWorkResult>,
386    aggregate_sender: mpsc::Sender<Proxy>,
387    aggregate_progress: ValidationProgress,
388    dead_map: Option<GroupDeadMap>,
389) {
390    let mut states: HashMap<GroupKey, GroupState> = HashMap::new();
391    while let Some(msg) = group_rx.recv().await {
392        let entry = states.entry(msg.key).or_insert_with(|| GroupState {
393            remaining: msg.group_len,
394            results: vec![None; msg.group_len],
395        });
396        entry.results[msg.slot] = msg.proxy;
397        entry.remaining -= 1;
398        if entry.remaining == 0 {
399            let finished = states
400                .remove(&msg.key)
401                .expect("current group state was pushed above");
402            if let Some(map) = dead_map.as_ref() {
403                map.lock()
404                    .unwrap_or_else(|e| e.into_inner())
405                    .remove(&msg.key);
406            }
407            aggregate_progress
408                .done
409                .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
410            if let Some(proxy) = group_finish(finished) {
411                aggregate_progress
412                    .passed
413                    .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
414                let _ = aggregate_sender.send(proxy).await;
415            }
416        }
417    }
418}
