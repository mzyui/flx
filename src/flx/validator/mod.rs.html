57fn validator_channel_capacity(concurrency_limit: usize) -> usize {
58    concurrency_limit
59        .saturating_mul(4)
60        .clamp(VALIDATOR_CHANNEL_MIN, VALIDATOR_CHANNEL_MAX)
61}

63/// Cooperative pause gate for validation workers.
64///
65/// Workers check [`PauseGate::wait_if_paused`] before starting each new
66/// probe; in-flight probes always run to completion. Clone the shared
67/// handle to drive pause state from elsewhere (e.g. a signal handler).
68#[derive(Debug, Default)]
69pub struct PauseGate {
70    paused: AtomicBool,
71    notify: Notify,
72}
73
74impl PauseGate {
75    /// Creates an unpaused gate.
76    pub fn new() -> Self {
77        Self::default()
78    }
79
80    /// Holds new probes; in-flight probes finish normally.
81    pub fn pause(&self) {
82        self.paused.store(true, Ordering::Relaxed);
83    }
84
85    /// Lets held probes start again.
86    pub fn resume(&self) {
87        self.paused.store(false, Ordering::Relaxed);
88        self.notify.notify_waiters();
89    }
90
91    /// Whether new probes are currently held.
92    pub fn is_paused(&self) -> bool {
93        self.paused.load(Ordering::Relaxed)
94    }
95
96    /// Waits while paused; returns immediately when running.
97    pub async fn wait_if_paused(&self) {
98        loop {
99            let notified = self.notify.notified();
100            tokio::pin!(notified);
101            notified.as_mut().enable();
102            if !self.is_paused() {
103                break;
104            }
105            notified.await;
106        }
107    }
108}

151const JUDGE_POOL_CACHE_TTL: Duration = Duration::from_secs(5 * 60);
152
153#[derive(Clone, PartialEq, Eq, Hash)]
154struct PoolKey {
155    urls: Vec<String>,
156    insecure: bool,
157}
158
159struct CachedJudgePool {
160    created: std::time::Instant,
161    pool: Arc<checker::JudgePool>,
162}
163
164static POOL_CACHE: std::sync::OnceLock<Mutex<HashMap<PoolKey, CachedJudgePool>>> =
165    std::sync::OnceLock::new();
166
167fn pool_cache() -> &'static Mutex<HashMap<PoolKey, CachedJudgePool>> {
168    POOL_CACHE.get_or_init(|| Mutex::new(HashMap::new()))
169}
170
171fn cached_judge_pool(urls: &[String], insecure: bool) -> Option<Arc<checker::JudgePool>> {
172    let mut cache = pool_cache().lock().unwrap_or_else(|e| e.into_inner());
173    cache.retain(|_, entry| entry.created.elapsed() <= JUDGE_POOL_CACHE_TTL);
174    let key = PoolKey {
175        urls: urls.to_vec(),
176        insecure,
177    };
178    cache.get(&key).map(|entry| Arc::clone(&entry.pool))
179}
180
181fn cache_judge_pool(urls: &[String], insecure: bool, pool: &Arc<checker::JudgePool>) {
182    let key = PoolKey {
183        urls: urls.to_vec(),
184        insecure,
185    };
186    pool_cache()
187        .lock()
188        .unwrap_or_else(|e| e.into_inner())
189        .insert(
190            key,
191            CachedJudgePool {
192                created: std::time::Instant::now(),
193                pool: Arc::clone(pool),
194            },
195        );
196}

198async fn preflight_pool(
199    urls: &[String],
200    timeout: Duration,
201    insecure: bool,
202) -> anyhow::Result<(Arc<checker::JudgePool>, JudgeHealthReport)> {
203    const PREFLIGHT_RETRIES: usize = 1;
204    const PREFLIGHT_RETRY_DELAY: Duration = Duration::from_secs(1);
205    if let Some(pool) = cached_judge_pool(urls, insecure) {
206        let report = JudgeHealthReport {
207            candidates: unique_count(urls),
208            healthy: pool.len(),
209            failed: Vec::new(),
210        };
211        return Ok((pool, report));
212    }
213    let mut last_error: Option<anyhow::Error> = None;
214    for attempt in 0..=PREFLIGHT_RETRIES {
215        if attempt > 0 {
216            tokio::time::sleep(PREFLIGHT_RETRY_DELAY).await;
217        }
218        let failed: Arc<Mutex<Vec<(String, String)>>> = Arc::new(Mutex::new(Vec::new()));
219        let failed_cb = Arc::clone(&failed);
220        match checker::JudgePool::build(urls, timeout, insecure, move |url, reason| {
221            report_dropped(url, reason);
222            failed_cb
223                .lock()
224                .unwrap_or_else(|e| e.into_inner())
225                .push((url.to_owned(), reason.to_owned()));
226        })
227        .await
228        {
229            Ok(pool) => {
230                let candidates = unique_count(urls);
231                let deadline = tokio::time::Instant::now() + timeout + Duration::from_secs(1);
232                let grace = tokio::time::Instant::now() + Duration::from_millis(250);
233                loop {
234                    let failed_len = failed.lock().unwrap_or_else(|e| e.into_inner()).len();
235                    if pool.len() + failed_len >= candidates
236                        || tokio::time::Instant::now() >= deadline
237                        || tokio::time::Instant::now() >= grace
238                    {
239                        break;
240                    }
241                    tokio::time::sleep(Duration::from_millis(25)).await;
242                }
243                let report = JudgeHealthReport {
244                    candidates,
245                    healthy: pool.len(),
246                    failed: failed.lock().unwrap_or_else(|e| e.into_inner()).clone(),
247                };
248                cache_judge_pool(urls, insecure, &pool);
249                return Ok((pool, report));
250            }
251            Err(error) => last_error = Some(error),
252        }
253    }
254    Err(last_error.expect("preflight_pool always runs at least one attempt"))
255}
256
257fn unique_count(urls: &[String]) -> usize {
258    urls.iter().collect::<std::collections::HashSet<_>>().len()
259}

261impl ProxyValidator {
262    /// Validates every proxy from the source stream against online judges.
263    ///
264    /// # Arguments
265    ///
266    /// * `proxy_source` - Candidate stream, usually from [`Flx`](crate::Flx).
267    /// * `config` - Protocols, timeouts, and concurrency limits.
268    ///
269    /// # Errors
270    ///
271    /// Returns an error when the config is empty/invalid or judge preflight fails.
272    pub async fn validate<S>(proxy_source: S, config: Config) -> anyhow::Result<Self>
273    where
274        S: Stream<Item = Proxy> + Send + 'static,
275    {
276        if config.types.is_empty() && config.groups.is_empty() {
277            anyhow::bail!("config.types and config.groups cannot both be empty; please specify at least one type.");
278        }
279        if config.groups.iter().any(|group| group.is_empty()) {
280            anyhow::bail!("config.groups cannot contain an empty group");
281        }
282        if config.concurrency_limit == 0 {
283            anyhow::bail!("config.concurrency_limit must be greater than zero");
284        }
285        if config.request_timeout == 0 {
286            anyhow::bail!("config.request_timeout must be greater than zero");
287        }
288        if config.max_attempts == 0 {
289            anyhow::bail!("config.max_attempts must be greater than zero");
290        }
291        #[cfg(feature = "log")]
292        log::debug!(
293            "Proxy validator started ({} workers)",
294            config.concurrency_limit
295        );
296
297        let (sender, receiver) =
298            mpsc::channel(validator_channel_capacity(config.concurrency_limit));
299        let (failure_tx, failure_rx) = if config.report_failures {
300            let (tx, rx) = mpsc::channel(validator_channel_capacity(config.concurrency_limit));
301            (Some(tx), Some(rx))
302        } else {
303            (None, None)
304        };
305        let progress = ValidationProgress::default();
306        let pause_gate = Arc::new(PauseGate::new());
307        let manager_pause = Arc::clone(&pause_gate);
308        let manager_total = Arc::clone(&progress.total);
309        let manager_done = Arc::clone(&progress.done);
310        let manager_passed = Arc::clone(&progress.passed);
311        let expected: Arc<[Protocol]> = Arc::from(config.types.into_boxed_slice());
312        let groups: Arc<Vec<Vec<Protocol>>> = Arc::new(
313            config
314                .groups
315                .into_iter()
316                .map(|mut group| {
317                    let mut seen: Vec<Protocol> = Vec::with_capacity(group.len());
318                    group.retain(|protocol| {
319                        if seen.contains(protocol) {
320                            false
321                        } else {
322                            seen.push(*protocol);
323                            true
324                        }
325                    });
326                    group
327                })
328                .collect(),
329        );
330        let group_spec: Arc<Vec<(usize, usize, Protocol)>> = Arc::from(
331            groups
332                .iter()
333                .enumerate()
334                .flat_map(|(group_idx, protocols)| {
335                    protocols
336                        .iter()
337                        .enumerate()
338                        .map(move |(slot, protocol)| (group_idx, slot, *protocol))
339                })
340                .collect::<Vec<_>>(),
341        );
342        let max_attempts = config.max_attempts;
343        let request_timeout = config.request_timeout;
344        let concurrency_limit = config.concurrency_limit;
345        let insecure = config.insecure;
346        let probe_missed = config.probe_missed_types;
347        let probe_gate = config.probe_gate.clone();
348        let support_cookies = config.support_cookies;
349        let support_referer = config.support_referer;
350        let preflight_timeout = Duration::from_secs(config.request_timeout);
351        let need_http = expected
352            .iter()
353            .chain(groups.iter().flatten())
354            .any(|protocol| matches!(protocol, Protocol::Http(_)));
355        let need_tunnel = expected
356            .iter()
357            .chain(groups.iter().flatten())
358            .any(|protocol| !matches!(protocol, Protocol::Http(_)));
359
360        let (buf_tx, buf_rx) = mpsc::channel(validator_channel_capacity(concurrency_limit));
361        let proxy_source: Pin<Box<dyn Stream<Item = Proxy> + Send>> = {
362            let mut src: Pin<Box<dyn Stream<Item = Proxy> + Send>> = Box::pin(proxy_source);
363            let tx = buf_tx;
364            tokio::spawn(async move {
365                while let Some(proxy) = src.next().await {
366                    if tx.send(proxy).await.is_err() {
367                        break;
368                    }
369                }
370            });
371            Box::pin(BufferedProxyStream { rx: buf_rx })
372        };
373
374        let http_preflight = async {
375            if !need_http {
376                return Ok::<Option<(Arc<checker::JudgePool>, JudgeHealthReport)>, anyhow::Error>(
377                    None,
378                );
379            }
380            let (pool, report) =
381                preflight_pool(&config.http_judge_urls, preflight_timeout, insecure)
382                    .await
383                    .context("HTTP online judge pool is empty after preflight")?;
384            Ok::<_, anyhow::Error>(Some((pool, report)))
385        };
386        let tunnel_preflight = async {
387            if !need_tunnel {
388                return Ok::<Option<(Arc<checker::JudgePool>, JudgeHealthReport)>, anyhow::Error>(
389                    None,
390                );
391            }
392            let (pool, report) =
393                preflight_pool(&config.https_judge_urls, preflight_timeout, insecure)
394                    .await
395                    .context("HTTPS online judge pool is empty after preflight")?;
396            Ok::<_, anyhow::Error>(Some((pool, report)))
397        };
398        tokio::spawn(async {
399            let _ = crate::resolver::my_ip().await;
400        });
401        let (http_target, tunnel_target) = tokio::join!(http_preflight, tunnel_preflight);
402        let http_target = http_target?;
403        let tunnel_target = tunnel_target?;
404        let mut judge_health = JudgeHealthReport::default();
405        if let Some((_, report)) = http_target.as_ref() {
406            judge_health.merge(report);
407        }
408        if let Some((_, report)) = tunnel_target.as_ref() {
409            judge_health.merge(report);
410        }
411        #[cfg(feature = "log")]
412        if let Some((pool, _)) = http_target.as_ref() {
413            log::info!("using {} healthy HTTP judge(s)", pool.len());
414        }
415        #[cfg(feature = "log")]
416        if let Some((pool, _)) = tunnel_target.as_ref() {
417            log::info!("using {} healthy HTTPS judge(s)", pool.len());
418        }
419        let targets = JudgeTargets {
420            http: http_target
421                .map(|(pool, _)| pool)
422                .unwrap_or_else(|| Arc::new(checker::JudgePool::from_targets(Vec::new()))),
423            tunnel: tunnel_target
424                .map(|(pool, _)| pool)
425                .unwrap_or_else(|| Arc::new(checker::JudgePool::from_targets(Vec::new()))),
426        };
427
428        let (group_tx, group_rx): (
429            mpsc::Sender<GroupWorkResult>,
430            mpsc::Receiver<GroupWorkResult>,
431        ) = mpsc::channel(validator_channel_capacity(concurrency_limit));
432
433        let aggregate_sender = sender.clone();
434        let aggregate_progress = progress.clone();
435        let worker_group_dead: work::GroupDeadMap =
436            std::sync::Arc::new(std::sync::Mutex::new(std::collections::HashMap::new()));
437        let group_aggregator = tokio::spawn(aggregate_groups(
438            group_rx,
439            aggregate_sender,
440            aggregate_progress,
441            Some(std::sync::Arc::clone(&worker_group_dead)),
442        ));
443
444        let manager = tokio::spawn(async move {
445            enum Job {
446                Singleton {
447                    proxy: Arc<Proxy>,
448                    protocol: Protocol,
449                    requested: Protocol,
450                },
451                GroupMember {
452                    proxy: Arc<Proxy>,
453                    proxy_id: u64,
454                    protocol: Protocol,
455                    group_idx: usize,
456                    slot: usize,
457                    group_len: usize,
458                },
459            }
460
461            let worker_counters = ValidationProgress {
462                total: Arc::clone(&manager_total),
463                done: Arc::clone(&manager_done),
464                passed: Arc::clone(&manager_passed),
465            };
466
467            let next_proxy_id = AtomicU64::new(0);
468            let stream_total = Arc::clone(&manager_total);
469            let jobs = proxy_source.flat_map(move |proxy: Proxy| {
470                let proxy = Arc::new(proxy);
471                let proxy_id = next_proxy_id.fetch_add(1, Ordering::Relaxed);
472                let advertised = Arc::clone(&proxy.expected_types);
473                let has_group = !group_spec.is_empty();
474
475                let singleton_jobs =
476                    work::singleton_jobs(advertised.as_ref(), expected.as_ref(), probe_missed);
477
478                let singleton_proxy = Arc::clone(&proxy);
479                let singleton: futures_util::stream::BoxStream<'static, Job> =
480                    Box::pin(futures_util::stream::iter(singleton_jobs.into_iter().map(
481                        move |(protocol, requested)| Job::Singleton {
482                            proxy: Arc::clone(&singleton_proxy),
483                            protocol,
484                            requested,
485                        },
486                    )));
487
488                let group: futures_util::stream::BoxStream<'static, Job> = if has_group {
489                    Box::pin(futures_util::stream::unfold(
490                        (
491                            0usize,
492                            proxy_id,
493                            proxy,
494                            Arc::clone(&group_spec),
495                            Arc::clone(&groups),
496                        ),
497                        |(mut idx, proxy_id, proxy, spec, groups)| async move {
498                            if idx >= spec.len() {
499                                return None;
500                            }
501                            let (group_idx, slot, protocol) = spec[idx];
502                            idx += 1;
503                            let group_len = groups[group_idx].len();
504                            Some((
505                                Job::GroupMember {
506                                    proxy: Arc::clone(&proxy),
507                                    proxy_id,
508                                    protocol,
509                                    group_idx,
510                                    slot,
511                                    group_len,
512                                },
513                                (idx, proxy_id, proxy, spec, groups),
514                            ))
515                        },
516                    ))
517                } else {
518                    Box::pin(futures_util::stream::empty())
519                };
520
521                singleton.chain(group).inspect({
522                    let stream_total = Arc::clone(&stream_total);
523                    move |job| match job {
524                        Job::Singleton { .. } => {
525                            stream_total.fetch_add(1, Ordering::Relaxed);
526                        }
527                        Job::GroupMember { slot, .. } if *slot == 0 => {
528                            stream_total.fetch_add(1, Ordering::Relaxed);
529                        }
530                        _ => {}
531                    }
532                })
533            });
534
535            let worker_group_tx = group_tx.clone();
536            let worker_failures = failure_tx.clone();
537            let worker_pause = Arc::clone(&manager_pause);
538            let worker_gate = probe_gate.clone();
539            jobs.for_each_concurrent(concurrency_limit, move |job| {
540                let sender = sender.clone();
541                let pause = worker_pause.clone();
542                let gate = worker_gate.clone();
543                let counters = worker_counters.clone();
544                let targets = targets.clone();
545                let group_tx = worker_group_tx.clone();
546                let failures = worker_failures.clone();
547                let group_dead = std::sync::Arc::clone(&worker_group_dead);
548                let params = WorkParams {
549                    max_attempts,
550                    request_timeout: Duration::from_secs(request_timeout),
551                    insecure,
552                    support_cookies,
553                    support_referer,
554                    retry_delay: config.retry_delay,
555                };
556                async move {
557                    pause.wait_if_paused().await;
558                    match job {
559                        Job::Singleton {
560                            proxy,
561                            protocol,
562                            requested,
563                        } => {
564                            if gate.as_ref().is_some_and(|gate| !gate(requested)) {
565                                counters.done.fetch_add(1, Ordering::Relaxed);
566                                return;
567                            }
568                            if let Err(_e) = do_work(
569                                SingletonJob {
570                                    proxy,
571                                    protocol,
572                                    requested,
573                                },
574                                sender,
575                                counters,
576                                targets,
577                                &params,
578                                failures,
579                            )
580                            .await
581                            {
582                                #[cfg(feature = "log")]
583                                log::debug!("validation task failed: {:#}", _e);
584                            }
585                        }
586                        Job::GroupMember {
587                            proxy,
588                            proxy_id,
589                            protocol,
590                            group_idx,
591                            slot,
592                            group_len,
593                        } => {
594                            if let Err(_e) = do_group_work(
595                                GroupMemberJob {
596                                    proxy,
597                                    proxy_id,
598                                    protocol,
599                                    group_idx,
600                                    slot,
601                                    group_len,
602                                },
603                                group_tx,
604                                targets,
605                                &params,
606                                failures,
607                                Some(group_dead),
608                            )
609                            .await
610                            {
611                                #[cfg(feature = "log")]
612                                log::debug!("group validation task failed: {:#}", _e);
613                            }
614                        }
615                    }
616                }
617            })
618            .await;
619
620            drop(group_tx);
621        });
622
623        Ok(Self {
624            receiver,
625            progress,
626            judge_health,
627            pause_gate,
628            #[cfg(feature = "log")]
629            timer: Instant::now(),
630            task_handle: manager,
631            group_task: group_aggregator,
632            failures: failure_rx,
633        })
634    }

641    /// Returns a shared handle to the live validation counters.
642    pub fn progress(&self) -> ValidationProgress {
643        self.progress.clone()
644    }
645
646    /// Pauses starting new probes; in-flight probes finish normally.
647    pub fn pause(&self) {
648        self.pause_gate.pause();
649    }
650
651    /// Resumes starting new probes after [`ProxyValidator::pause`].
652    pub fn resume(&self) {
653        self.pause_gate.resume();
654    }
655
656    /// Whether new probes are currently held.
657    pub fn is_paused(&self) -> bool {
658        self.pause_gate.is_paused()
659    }
660
661    /// Shares the pause gate (e.g. with a signal handler task).
662    pub fn pause_gate(&self) -> Arc<PauseGate> {
663        Arc::clone(&self.pause_gate)
664    }
665
666    /// Takes the failure-report receiver; returns [`None`] when reporting is
667    /// disabled or the receiver was already taken. Call before draining the
668    /// stream so no buffered failure records are dropped.
669    pub fn take_failures(&mut self) -> Option<mpsc::Receiver<work::ProxyFailure>> {
670        self.failures.take()
671    }

673    /// Receives the next passing proxy, or [`None`] once validation is complete
674    /// and all passing proxies have been consumed.

676        self.receiver.recv().await
677    }

688impl Drop for ProxyValidator {
689    fn drop(&mut self) {
690        self.receiver.close();
691        self.task_handle.abort();
692        self.group_task.abort();
693        #[cfg(feature = "log")]
694        log::info!(

1192    async fn spawn_mock_http_proxy() -> u16 {
1193        use tokio::io::{AsyncReadExt as _, AsyncWriteExt as _};
1194        use tokio::net::TcpListener;
1195
1196        let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
1197        let port = listener.local_addr().unwrap().port();
1198        tokio::spawn(async move {
1199            loop {
1200                let Ok((mut stream, _)) = listener.accept().await else {
1201                    return;
1202                };
1203                tokio::spawn(async move {
1204                    let mut buf = Vec::new();
1205                    let mut chunk = [0u8; 2048];
1206                    loop {
1207                        let read = stream.read(&mut chunk).await.unwrap_or(0);
1208                        if read == 0 {
1209                            break;
1210                        }
1211                        buf.extend_from_slice(&chunk[..read]);
1212                        if buf.windows(4).any(|w| w == b"\r\n\r\n") {
1213                            break;
1214                        }
1215                    }
1216                    let text = String::from_utf8_lossy(&buf);
1217                    let token = text
1218                        .lines()
1219                        .find_map(|line| {
1220                            let (name, value) = line.split_once(':')?;
1221                            name.eq_ignore_ascii_case("x-fluxy-token")
1222                                .then(|| value.trim().to_owned())
1223                        })
1224                        .unwrap_or_default();
1225                    let body = format!("HTTP_X_FLUXY_TOKEN = {token}\n");
1226                    let response = format!(
1227                        "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{}",
1228                        body.len(),
1229                        body
1230                    );
1231                    let _ = stream.write_all(response.as_bytes()).await;
1232                });
1233            }
1234        });
1235        port
