29/// DNS sources for public-IP discovery: (server, domain, qtype).
30///

32/// `o-o.myaddr.l.google.com` answers TXT with it. Both are queried directly

64async fn load_cached_public_ip(path: &Path) -> Option<String> {
65    let modified = tokio::fs::metadata(path).await.ok()?.modified().ok()?;
66    let age = SystemTime::now().duration_since(modified).ok()?;
67    if age > PUBLIC_IP_CACHE_TTL {
68        return None;
69    }
70    let body = tokio::fs::read_to_string(path).await.ok()?;
71    let ip = body.trim();
72    if ip.parse::<IpAddr>().is_ok() {
73        Some(ip.to_string())
74    } else {
75        None
76    }
77}

79async fn store_public_ip(path: &Path, ip: &str) {
80    if let Some(parent) = path.parent() {
81        let _ = tokio::fs::create_dir_all(parent).await;
82    }
83    let name = path
84        .file_name()
85        .and_then(|name| name.to_str())
86        .unwrap_or("public-ip");
87    let tmp = path.with_file_name(format!(".{name}.tmp-{}", std::process::id()));
88    if tokio::fs::write(&tmp, format!("{ip}\n")).await.is_err() {
89        return;
90    }
91    if tokio::fs::rename(&tmp, path).await.is_err() {
92        let _ = tokio::fs::remove_file(&tmp).await;
93    }
94}

96fn parse_ip_body(body: &[u8]) -> anyhow::Result<String> {
97    if body.len() > MAX_IP_BODY_BYTES {
98        anyhow::bail!("public-IP response exceeds {MAX_IP_BODY_BYTES} bytes");
99    }
100    let text = std::str::from_utf8(body).context("public-IP response is not valid UTF-8")?;
101    let ip: IpAddr = text
102        .trim()
103        .parse()
104        .with_context(|| format!("endpoint did not return a valid IP: {text:?}"))?;
105    Ok(ip.to_string())
106}

108async fn fetch_ip_endpoint(
109    client: Client<crate::proxy::client::HttpsConnector, Empty<Bytes>>,
110    endpoint: &'static str,
111    deadline: time::Instant,
112) -> anyhow::Result<String> {
113    let response = time::timeout_at(deadline, client.get(endpoint.parse()?))
114        .await
115        .with_context(|| format!("request to {endpoint} timed out"))?
116        .with_context(|| format!("request to {endpoint} failed"))?;
117    if !response.status().is_success() {
118        anyhow::bail!("{endpoint} returned status {}", response.status());
119    }
120
121    let mut body = response.into_body();
122    let mut bytes = Vec::with_capacity(32);
123    while let Some(frame) = time::timeout_at(deadline, body.frame())
124        .await
125        .with_context(|| format!("body from {endpoint} timed out"))?
126    {
127        let chunk = frame
128            .with_context(|| format!("body from {endpoint} was interrupted"))?
129            .into_data()
130            .unwrap_or_default();
131        if bytes.len().saturating_add(chunk.len()) > MAX_IP_BODY_BYTES {
132            anyhow::bail!("{endpoint} response exceeds {MAX_IP_BODY_BYTES} bytes");
133        }
134        bytes.extend_from_slice(&chunk);
135    }
136    parse_ip_body(&bytes).with_context(|| format!("invalid response from {endpoint}"))
137}

139/// Resolves the public IP via DNS, racing every [`DNS_IP_SOURCES`] entry.
140///
141/// No sticky kill-switch: a transient failure only fails this call, the next
142/// one retries all sources fresh.
143async fn my_ip_via_dns() -> anyhow::Result<String> {
144    let mut pending = DNS_IP_SOURCES
145        .into_iter()
146        .map(|(server, domain, qtype)| dns_ip_lookup(server, domain, qtype))
147        .collect::<FuturesUnordered<_>>();
148    let mut errors = Vec::new();
149    while let Some(result) = pending.next().await {
150        match result {
151            Ok(ip) => return Ok(ip),
152            Err(error) => errors.push(format!("{error:#}")),
153        }
154    }
155    anyhow::bail!("all DNS public-IP sources failed: {}", errors.join("; "))
156}

158static DNS_QUERY_ID: AtomicU16 = AtomicU16::new(0);

160fn next_dns_query_id() -> u16 {
161    let nanos = SystemTime::now()
162        .duration_since(SystemTime::UNIX_EPOCH)
163        .map(|elapsed| elapsed.subsec_nanos() as u16)
164        .unwrap_or(0);
165    DNS_QUERY_ID.fetch_add(1, Ordering::Relaxed) ^ nanos
166}

168async fn dns_ip_lookup(server: Ipv4Addr, domain: &str, qtype: u16) -> anyhow::Result<String> {
169    let query_id = next_dns_query_id();
170    let mut query = [0u8; DNS_QUERY_BUFFER_LEN];
171    let query_len = build_dns_query(domain, qtype, query_id, &mut query)?;
172
173    let socket = tokio::net::UdpSocket::bind((Ipv4Addr::UNSPECIFIED, 0))
174        .await
175        .context("failed to bind UDP socket for DNS lookup")?;
176    socket
177        .send_to(&query[..query_len], SocketAddr::from((server, DNS_PORT)))
178        .await
179        .context("failed to send DNS query")?;
180
181    let mut response = [0u8; DNS_RESPONSE_BUFFER_LEN];
182    let response_len = time::timeout(LOOKUP_TIMEOUT, socket.recv(&mut response))
183        .await
184        .context("DNS lookup timed out")?
185        .context("DNS recv failed")?;
186    let message = &response[..response_len];
187    match qtype {
188        DNS_TYPE_A => parse_dns_a_response(message, query_id).map(|ip| ip.to_string()),
189        DNS_TYPE_TXT => parse_dns_txt_response(message, query_id),
190        other => anyhow::bail!("unsupported DNS query type {other} for `{domain}`"),
191    }
192    .with_context(|| format!("invalid DNS response for `{domain}`"))
193}

195fn build_dns_query(
196    domain: &str,
197    qtype: u16,
198    query_id: u16,
199    out: &mut [u8; DNS_QUERY_BUFFER_LEN],
200) -> anyhow::Result<usize> {
201    out[..2].copy_from_slice(&query_id.to_be_bytes());
202    out[2..4].copy_from_slice(&DNS_QUERY_FLAGS.to_be_bytes());
203    out[4..6].copy_from_slice(&1u16.to_be_bytes());
204
205    let mut len = DNS_HEADER_LEN;
206    let mut labels = 0usize;
207    for label in domain.split('.') {
208        let label = label.as_bytes();
209        if label.is_empty() {
210            anyhow::bail!("empty DNS label in `{domain}`");
211        }
212        if label.len() > DNS_MAX_LABEL_LEN {
213            anyhow::bail!("DNS label exceeds {DNS_MAX_LABEL_LEN} bytes in `{domain}`");
214        }
215        let end = len + 1 + label.len();
216        if end + 1 + DNS_QUESTION_TAIL_LEN > out.len() {
217            anyhow::bail!("DNS name `{domain}` exceeds the {DNS_MAX_NAME_LEN}-byte wire limit");
218        }
219        out[len] = label.len() as u8;
220        out[len + 1..end].copy_from_slice(label);
221        len = end;
222        labels += 1;
223    }
224    if labels == 0 {
225        anyhow::bail!("empty DNS name");
226    }
227    out[len] = 0;
228    len += 1;
229    out[len..len + 2].copy_from_slice(&qtype.to_be_bytes());
230    out[len + 2..len + 4].copy_from_slice(&DNS_CLASS_IN.to_be_bytes());
231    Ok(len + DNS_QUESTION_TAIL_LEN)
232}

234fn parse_dns_a_response(message: &[u8], query_id: u16) -> anyhow::Result<Ipv4Addr> {
235    if message.len() < DNS_HEADER_LEN {
236        anyhow::bail!("DNS response is shorter than its header");
237    }
238    let id = u16::from_be_bytes([message[0], message[1]]);
239    if id != query_id {
240        anyhow::bail!("DNS response id {id} does not match query id {query_id}");
241    }
242    let flags = u16::from_be_bytes([message[2], message[3]]);
243    if flags & DNS_FLAG_QR == 0 {
244        anyhow::bail!("DNS reply is not a response");
245    }
246    if flags & DNS_FLAG_TC != 0 {
247        anyhow::bail!("DNS response was truncated");
248    }
249    let rcode = flags & DNS_RCODE_MASK;
250    if rcode != 0 {
251        anyhow::bail!("DNS server returned rcode {rcode}");
252    }
253    let question_count = u16::from_be_bytes([message[4], message[5]]);
254    let answer_count = u16::from_be_bytes([message[6], message[7]]);
255
256    let mut offset = DNS_HEADER_LEN;
257    for _ in 0..question_count {
258        offset = skip_dns_name(message, offset).context("malformed name in question")?;
259        offset += DNS_QUESTION_TAIL_LEN;
260        if offset > message.len() {
261            anyhow::bail!("question section overruns the response");
262        }
263    }
264    for _ in 0..answer_count {
265        offset = skip_dns_name(message, offset).context("malformed name in answer")?;
266        let Some(record) = message.get(offset..offset + DNS_ANSWER_FIXED_LEN) else {
267            anyhow::bail!("answer header overruns the response");
268        };
269        let record_type = u16::from_be_bytes([record[0], record[1]]);
270        let rdlength = u16::from_be_bytes([record[8], record[9]]) as usize;
271        let rdata = offset + DNS_ANSWER_FIXED_LEN;
272        let rdata_end = rdata + rdlength;
273        if rdata_end > message.len() {
274            anyhow::bail!("answer record overruns the response");
275        }
276        if record_type == DNS_TYPE_A && rdlength == 4 {
277            return Ok(Ipv4Addr::new(
278                message[rdata],
279                message[rdata + 1],
280                message[rdata + 2],
281                message[rdata + 3],
282            ));
283        }
284        offset = rdata_end;
285    }
286    anyhow::bail!("DNS response carried no A record")
287}

289fn skip_dns_name(message: &[u8], mut offset: usize) -> Option<usize> {
290    loop {
291        let length = *message.get(offset)? as usize;
292        if length & 0xC0 == 0xC0 {
293            return offset.checked_add(2);
294        }
295        if length == 0 {
296            return Some(offset + 1);
297        }
298        offset += 1 + length;
299    }
300}

302/// Validates the header and returns the answer section offset plus count.
303fn answers_offset(message: &[u8], query_id: u16) -> anyhow::Result<(usize, u16)> {
304    if message.len() < DNS_HEADER_LEN {
305        anyhow::bail!("DNS response is shorter than its header");
306    }
307    let id = u16::from_be_bytes([message[0], message[1]]);
308    if id != query_id {
309        anyhow::bail!("DNS response id {id} does not match query id {query_id}");
310    }
311    let flags = u16::from_be_bytes([message[2], message[3]]);
312    if flags & DNS_FLAG_QR == 0 {
313        anyhow::bail!("DNS reply is not a response");
314    }
315    if flags & DNS_FLAG_TC != 0 {
316        anyhow::bail!("DNS response was truncated");
317    }
318    let rcode = flags & DNS_RCODE_MASK;
319    if rcode != 0 {
320        anyhow::bail!("DNS server returned rcode {rcode}");
321    }
322    let question_count = u16::from_be_bytes([message[4], message[5]]);
323    let answer_count = u16::from_be_bytes([message[6], message[7]]);
324
325    let mut offset = DNS_HEADER_LEN;
326    for _ in 0..question_count {
327        offset = skip_dns_name(message, offset).context("malformed name in question")?;
328        offset += DNS_QUESTION_TAIL_LEN;
329        if offset > message.len() {
330            anyhow::bail!("question section overruns the response");
331        }
332    }
333    Ok((offset, answer_count))
334}

336/// Reads the first TXT record as the public IP (Google's `o-o.myaddr.l.google.com`).
337fn parse_dns_txt_response(message: &[u8], query_id: u16) -> anyhow::Result<String> {
338    let (mut offset, answer_count) = answers_offset(message, query_id)?;
339    for _ in 0..answer_count {
340        offset = skip_dns_name(message, offset).context("malformed name in answer")?;
341        let Some(record) = message.get(offset..offset + DNS_ANSWER_FIXED_LEN) else {
342            anyhow::bail!("answer header overruns the response");
343        };
344        let record_type = u16::from_be_bytes([record[0], record[1]]);
345        let rdlength = u16::from_be_bytes([record[8], record[9]]) as usize;
346        let rdata = offset + DNS_ANSWER_FIXED_LEN;
347        let rdata_end = rdata + rdlength;
348        if rdata_end > message.len() {
349            anyhow::bail!("answer record overruns the response");
350        }
351        if record_type == DNS_TYPE_TXT && rdlength >= 1 {
352            let text_len = message[rdata] as usize;
353            let text_end = rdata + 1 + text_len;
354            if text_end <= rdata_end {
355                let text = std::str::from_utf8(&message[rdata + 1..text_end])
356                    .context("TXT record is not valid UTF-8")?;
357                let ip: std::net::IpAddr = text
358                    .trim()
359                    .parse()
360                    .with_context(|| format!("TXT record is not a valid IP: {text:?}"))?;
361                return Ok(ip.to_string());
362            }
363        }
364        offset = rdata_end;
365    }
366    anyhow::bail!("DNS response carried no TXT record")
367}

369async fn my_ip_via_https() -> anyhow::Result<String> {
370    let client = Client::builder(TokioExecutor::new())
371        .build::<_, Empty<Bytes>>(crate::proxy::client::https_connector());
372    let deadline = time::Instant::now() + LOOKUP_TIMEOUT;
373    let mut requests = HTTP_IP_ENDPOINTS
374        .into_iter()
375        .map(|endpoint| fetch_ip_endpoint(client.clone(), endpoint, deadline))
376        .collect::<FuturesUnordered<_>>();
377    let mut errors = Vec::new();
378    while let Some(result) = requests.next().await {
379        match result {
380            Ok(ip) => return Ok(ip),
381            Err(error) => errors.push(format!("{error:#}")),
382        }
383    }
384    anyhow::bail!("all HTTPS IP endpoints failed: {}", errors.join("; "))
385}

394/// Resolves the public IP, caching successes for the process lifetime.
395///
396/// Races DNS plus HTTPS endpoints within a fixed budget.
397///
398/// # Errors
399///
400/// Returns an error when every source fails or the lookup times out.
401pub async fn my_ip() -> anyhow::Result<String> {
402    MY_IP_CACHE
403        .get_or_try_init(|| async {
404            time::timeout(MY_IP_LOOKUP_TIMEOUT, resolve_public_ip())
405                .await
406                .context("public-IP lookup timed out")?
407        })
408        .await
409        .cloned()
410}

412async fn resolve_public_ip() -> anyhow::Result<String> {
413    let start_time = Instant::now();
414
415    let live = race_live_ip_sources().await;
416    let ip = match live {
417        Ok(ip) => {
418            #[cfg(feature = "log")]
419            log::debug!(
420                "My IP: {} (resolved via live sources in {:?})",
421                ip,
422                start_time.elapsed()
423            );
424            ip
425        }
426        Err(live_error) => {
427            if let Some(path) = public_ip_cache_path() {
428                if let Some(cached) = load_cached_public_ip(&path).await {
429                    #[cfg(feature = "log")]
430                    log::warn!(
431                        "live public-IP lookup failed ({live_error:#}); using cached IP {cached} as last resort"
432                    );
433                    #[cfg(not(feature = "log"))]
434                    let _ = &live_error;
435                    return Ok(cached);
436                }
437            }
438            return Err(live_error
439                .context("all live public-IP sources failed and no cached IP is available"));
440        }
441    };
442
443    if let Some(path) = public_ip_cache_path() {
444        store_public_ip(&path, &ip).await;
445    }
446    let _ = start_time;
447    Ok(ip)
448}

450async fn race_live_ip_sources() -> anyhow::Result<String> {
451    let mut pending = FuturesUnordered::new();
452    pending.push(async { my_ip_via_dns().await }.boxed());
453    pending.push(async { my_ip_via_https().await }.boxed());
454    let mut errors = Vec::new();
455    while let Some(result) = pending.next().await {
456        match result {
457            Ok(ip) => return Ok(ip),
458            Err(error) => errors.push(format!("{error:#}")),
459        }
460    }
461    anyhow::bail!("all live public-IP sources failed: {}", errors.join("; "))
462}

569        assert_eq!(DNS_QUERY_FLAGS & DNS_FLAG_QR, 0);

572        assert_eq!(&buffer[0..2], &0x1234u16.to_be_bytes());
573        assert_eq!(&buffer[2..4], &DNS_QUERY_FLAGS.to_be_bytes());
574        assert_eq!(&buffer[4..6], &1u16.to_be_bytes());
575        let expected_name = [
576            6, b'w', b'h', b'o', b'a', b'm', b'i', 6, b'a', b'k', b'a', b'm', b'a', b'i', 3, b'n',
577            b'e', b't', 0,
578        ];
579        assert_eq!(
580            &buffer[DNS_HEADER_LEN..len - DNS_QUESTION_TAIL_LEN],
581            &expected_name
582        );
583        let tail = &buffer[len - DNS_QUESTION_TAIL_LEN..len];
584        assert_eq!(&tail[..2], &DNS_TYPE_A.to_be_bytes());
585        assert_eq!(&tail[2..], &DNS_CLASS_IN.to_be_bytes());

701        let name = sample_question_name();
702        let mut message = Vec::new();
