42/// Track in-flight GeoLite2 download progress.
43#[derive(Debug, Clone, Copy)]
44pub struct DownloadProgress {
45    /// Database file being downloaded (`GeoLite2-City.mmdb` or ASN).
46    pub name: &'static str,
47    /// Bytes written so far.
48    pub downloaded: usize,
49    /// Total bytes from `Content-Length`, or 0 when unknown.
50    pub total: usize,
51}
52
53static DOWNLOAD_EVENTS: OnceLock<watch::Sender<Option<DownloadProgress>>> = OnceLock::new();
54
55/// Install a download observer and return its receiver.
56pub fn install_download_observer() -> Option<watch::Receiver<Option<DownloadProgress>>> {
57    let (tx, rx) = watch::channel(None);
58    match DOWNLOAD_EVENTS.set(tx) {
59        Ok(()) => Some(rx),
60        Err(tx) => {
61            let _ = tx.send(None);
62            None
63        }
64    }
65}
66
67fn report_download(event: Option<DownloadProgress>) {
68    if let Some(tx) = DOWNLOAD_EVENTS.get() {
69        let _ = tx.send_replace(event);
70    }
71}

113#[cfg(unix)]
114fn lock_file(file: &std::fs::File) -> std::io::Result<()> {
115    use std::os::fd::AsRawFd as _;
116
117    // Rely on flock releasing when the descriptor closes.
118    loop {
119        let rc = unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX) };
120        if rc == 0 {
121            return Ok(());
122        }
123        let error = std::io::Error::last_os_error();
124        if error.kind() != std::io::ErrorKind::Interrupted {
125            return Err(error);
126        }
127    }
128}

130#[cfg(not(unix))]
131fn lock_file(_file: &std::fs::File) -> std::io::Result<()> {
132    // Serialize non-unix runs via the in-process tokio mutex.
133    Ok(())
134}

136async fn acquire_download_lock(path: &Path) -> anyhow::Result<std::fs::File> {
137    let lock = lock_path(path);
138    // Bound lock waits so a wedged holder cannot block forever.
139    let acquire = tokio::task::spawn_blocking({
140        let lock = lock.clone();
141        move || {
142            let file = std::fs::OpenOptions::new()
143                .create(true)
144                .truncate(false)
145                .write(true)
146                .open(&lock)
147                .with_context(|| format!("failed to create {}", lock.display()))?;
148            lock_file(&file).with_context(|| format!("failed to lock {}", lock.display()))?;
149            Ok(file)
150        }
151    });
152    tokio::time::timeout(Duration::from_secs(5), acquire)
153        .await
154        .context("timed out waiting for the GeoLite2 database lock (held by another flx instance)")?
155        .context("GeoLite2 lock task failed")?
156}

158async fn write_database_chunks<S, E, P>(
159    mmdb_path: &Path,
160    chunks: S,
161    max_size: usize,
162    deadline: tokio::time::Instant,
163    mut progress: P,
164) -> anyhow::Result<()>
165where
166    S: Stream<Item = Result<Bytes, E>>,
167    E: std::error::Error + Send + Sync + 'static,
168    P: FnMut(usize),
169{
170    let temporary_path = temporary_path(mmdb_path);
171    let _lock = acquire_download_lock(mmdb_path).await?;
172    let result = async {
173        let mut file = tokio::fs::OpenOptions::new()
174            .create(true)
175            .truncate(true)
176            .write(true)
177            .open(&temporary_path)
178            .await
179            .with_context(|| format!("failed to create {}", temporary_path.display()))?;
180        let mut downloaded = 0usize;
181        let mut chunks = std::pin::pin!(chunks);
182
183        while let Some(next) = tokio::time::timeout_at(deadline, chunks.next())
184            .await
185            .context("GeoLite2 download body timed out")?
186        {
187            let chunk = next.context("GeoLite2 download stream interrupted")?;
188            downloaded = downloaded
189                .checked_add(chunk.len())
190                .context("GeoLite2 database size overflow")?;
191            if downloaded > max_size {
192                anyhow::bail!(
193                    "GeoLite2 database exceeds maximum size of {} bytes",
194                    max_size
195                );
196            }
197            progress(chunk.len());
198            file.write_all(&chunk)
199                .await
200                .with_context(|| format!("failed to write to {}", temporary_path.display()))?;
201        }
202
203        file.sync_all()
204            .await
205            .with_context(|| format!("failed to flush {}", temporary_path.display()))?;
206        drop(file);
207        tokio::task::spawn_blocking({
208            let temporary_path = temporary_path.clone();
209            move || Reader::open_readfile(&temporary_path).map(|_| ())
210        })
211        .await
212        .context("GeoLite2 validation task failed")?
213        .context("downloaded GeoLite2 file is not a valid MMDB database")?;
214        tokio::fs::rename(&temporary_path, mmdb_path)
215            .await
216            .with_context(|| {
217                format!(
218                    "failed to atomically install {} as {}",
219                    temporary_path.display(),
220                    mmdb_path.display()
221                )
222            })
223    }
224    .await;
225
226    if result.is_err() {
227        match tokio::fs::remove_file(&temporary_path).await {
228            Ok(()) => {}
229            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
230            Err(error) => {
231                return Err(error).with_context(|| {
232                    format!("failed to remove partial file {}", temporary_path.display())
233                });
234            }
235        }
236    }
237
238    result
239}

241/// Returns the `<data_dir>/flx` directory, creating it when missing.
242///
243/// Falls back to `./flx` under the current directory when no platform data
244/// directory exists.
245///
246/// # Errors
247///
248/// Returns an error when the directory cannot be determined or created.
249pub fn data_dir() -> anyhow::Result<PathBuf> {
250    if let Some(base) = crate::base_dirs::data_dir() {
251        let mut dir = base;
252        dir.push(env!("CARGO_PKG_NAME"));
253
254        if !dir.is_dir() {
255            fs::create_dir_all(&dir)
256                .with_context(|| format!("failed to create data directory {}", dir.display()))?;
257        }
258        Ok(dir)
259    } else {
260        #[cfg(feature = "log")]
261        log::warn!("Failed to get local data directory, using current directory instead");
262        let mut dir = current_dir().context("failed to determine current working directory")?;
263        dir.push(env!("CARGO_PKG_NAME"));
264        if !dir.is_dir() {
265            fs::create_dir_all(&dir).with_context(|| {
266                format!(
267                    "failed to create data directory {} under the current working directory (no platform data directory available); choose a writable working directory or set XDG_DATA_HOME",
268                    dir.display()
269                )
270            })?;
271        }
272        Ok(dir)
273    }
274}

290/// Relocate legacy databases so existing installs skip re-download.
291fn migrate_legacy_database(new_path: &Path) {
292    let Some(file_name) = new_path.file_name() else {
293        return;
294    };
295    let Some(legacy) = new_path
296        .parent()
297        .and_then(Path::parent)
298        .map(|dir| dir.join(file_name))
299    else {
300        return;
301    };
302    if new_path.exists() || !legacy.exists() {
303        return;
304    }
305    if let Err(error) = fs::rename(legacy, new_path) {
306        #[cfg(feature = "log")]
307        log::warn!(
308            "failed to move legacy database to {}: {error}",
309            new_path.display()
310        );
311        #[cfg(not(feature = "log"))]
312        let _ = error;
313        return;
314    }
315    // Move the ETag sidecar so the copy stays verifiable.
316    let marker = sync_marker_path(new_path);
317    if let Some(legacy_marker) = marker
318        .parent()
319        .and_then(Path::parent)
320        .map(|dir| dir.join(marker.file_name().unwrap_or_default()))
321    {
322        if !marker.exists() && legacy_marker.exists() {
323            let _ = fs::rename(legacy_marker, &marker);
324        }
325    }
326}

344fn read_synced_etag(mmdb_path: &Path) -> Option<String> {
345    let content = fs::read(sync_marker_path(mmdb_path)).ok()?;
346    let etag = std::str::from_utf8(&content).ok()?.trim();
347    (!etag.is_empty()).then(|| etag.to_owned())

350async fn write_synced_etag(mmdb_path: &Path, etag: &str) -> anyhow::Result<()> {
351    let marker = sync_marker_path(mmdb_path);
352    let temporary = temporary_path(&marker);
353    tokio::fs::write(&temporary, etag.as_bytes())
354        .await
355        .with_context(|| format!("failed to write {}", temporary.display()))?;
356    tokio::fs::rename(&temporary, &marker)
357        .await
358        .with_context(|| format!("failed to install {}", marker.display()))
359}

361async fn fetch_database(
362    url: &str,
363    if_none_match: Option<&str>,
364    deadline: tokio::time::Instant,
365) -> anyhow::Result<(hyper::Response<hyper::body::Incoming>, Option<String>)> {
366    let https_connector = crate::proxy::client::https_connector();
367    let client = Client::builder(TokioExecutor::new()).build(https_connector);
368
369    let mut builder = Request::builder().uri(url).header(
370        hyper::header::USER_AGENT,
371        crate::user_agent::next_user_agent(),
372    );
373    if let Some(etag) = if_none_match {
374        builder = builder.header(hyper::header::IF_NONE_MATCH, etag);
375    }
376    let req = builder
377        .body(Empty::<Bytes>::new())
378        .context("failed to build GeoLite2 download request")?;
379
380    // Bound connects separately so unreachable mirrors fail fast.
381    let request_deadline = deadline.min(tokio::time::Instant::now() + GEOLITE_CONNECT_TIMEOUT);
382    let response = tokio::time::timeout_at(request_deadline, client.request(req))
383        .await
384        .context("cannot reach the GeoLite2 mirror")?
385        .with_context(|| format!("failed to download GeoLite2 database from {url}"))?;
386    let etag = response
387        .headers()
388        .get(hyper::header::ETAG)
389        .and_then(|value| value.to_str().ok())
390        .map(ToOwned::to_owned);
391    Ok((response, etag))
392}

394async fn install_response(
395    response: hyper::Response<hyper::body::Incoming>,
396    mmdb_path: &Path,
397    url: &str,
398    deadline: tokio::time::Instant,
399) -> anyhow::Result<()> {
400    if !response.status().is_success() {
401        anyhow::bail!(
402            "GeoLite2 download from {} returned status {}",
403            url,
404            response.status()
405        );
406    }
407
408    let content_length = response
409        .headers()
410        .get(hyper::header::CONTENT_LENGTH)
411        .and_then(|value| value.to_str().ok())
412        .and_then(|value| value.parse::<usize>().ok());
413    if content_length.is_some_and(|length| length > MAX_DATABASE_SIZE) {
414        anyhow::bail!(
415            "GeoLite2 database exceeds maximum size of {} bytes",
416            MAX_DATABASE_SIZE
417        );
418    }
419
420    let name = database_name(url);
421    let total = content_length.unwrap_or(0);
422    let mut downloaded = 0usize;
423
424    let chunks = futures_util::stream::unfold(response, |mut response| async move {
425        loop {
426            match response.frame().await {
427                Some(Ok(frame)) => {
428                    if let Ok(chunk) = frame.into_data() {
429                        return Some((Ok(chunk), response));
430                    }
431                }
432                Some(Err(error)) => return Some((Err(error), response)),
433                None => return None,
434            }
435        }
436    });
437    write_database_chunks(
438        mmdb_path,
439        chunks,
440        MAX_DATABASE_SIZE,
441        deadline,
442        |chunk_size| {
443            downloaded = downloaded.saturating_add(chunk_size);
444            report_download(Some(DownloadProgress {
445                name,
446                downloaded,
447                total,
448            }));
449        },
450    )
451    .await
452}

462/// Outcome of syncing the local GeoLite2 databases with the mirror.
463#[derive(Debug, Clone, Copy, PartialEq, Eq)]
464pub enum SyncOutcome {
465    /// Local databases already match the mirror revision.
466    UpToDate,
467    /// At least one database was downloaded and installed.
468    Synced,
469}
470
471/// Syncs local GeoLite2 databases with the mirror revision.
472///
473/// Skips download when the ETag matches; migrates legacy layouts first.
474///
475/// # Errors
476///
477/// Returns an error when the data dir is unavailable or download fails.
478pub async fn sync_database() -> anyhow::Result<SyncOutcome> {
479    let city = sync_one(GEOLITE_ENDPOINT_URL, &database_path()?).await?;
480    let asn = sync_one(GEOLITE_ASN_ENDPOINT_URL, &asn_database_path()?).await?;
481    Ok(match (city, asn) {
482        (SyncOutcome::Synced, _) | (_, SyncOutcome::Synced) => SyncOutcome::Synced,
483        _ => SyncOutcome::UpToDate,
484    })
485}
486
487async fn sync_one(url: &str, mmdb_path: &Path) -> anyhow::Result<SyncOutcome> {
488    let db_valid = local_build_epoch(mmdb_path).is_some();
489    let stored_etag = read_synced_etag(mmdb_path);
490    let deadline = tokio::time::Instant::now() + DATABASE_DOWNLOAD_TIMEOUT;
491
492    // Skip conditional requests without a valid copy and ETag.
493    let conditional = db_valid && stored_etag.is_some();
494    let (response, remote_etag) = if conditional {
495        fetch_database(url, stored_etag.as_deref(), deadline).await?
496    } else {
497        fetch_database(url, None, deadline).await?
498    };
499
500    if response.status() == hyper::StatusCode::NOT_MODIFIED {
501        return Ok(SyncOutcome::UpToDate);
502    }
503
504    let _notifier = begin_download(database_name(url));
505    install_response(response, mmdb_path, url, deadline).await?;
506    if let Some(etag) = remote_etag {
507        write_synced_etag(mmdb_path, &etag).await?;
508    }
509    Ok(SyncOutcome::Synced)
510}

512/// Download a GeoLite2 database from the mirror into `mmdb_path`.
513pub async fn download_database(mmdb_path: &Path, url: &str) -> anyhow::Result<()> {
514    let deadline = tokio::time::Instant::now() + DATABASE_DOWNLOAD_TIMEOUT;
515    let _notifier = begin_download(database_name(url));
516    let (response, _etag) = fetch_database(url, None, deadline).await?;
517    install_response(response, mmdb_path, url, deadline).await
518}

520async fn ensure_database(
521    mmdb_path: &Path,
522    url: &str,
523    display_name: &str,
524) -> anyhow::Result<Reader<Vec<u8>>> {
525    if !mmdb_path.exists() {
526        #[cfg(feature = "log")]
527        log::debug!("{} does not exist, downloading", mmdb_path.display());
528        download_database(mmdb_path, url)
529            .await
530            .with_context(|| format!("failed to download {display_name}"))?;
531    }
532
533    match Reader::open_readfile(mmdb_path) {
534        Ok(reader) => Ok(reader),
535        Err(e) => {
536            // Drop corrupt files so the next run re-downloads.
537            if let Err(_remove_err) = remove_file(mmdb_path) {
538                #[cfg(feature = "log")]
539                log::warn!(
540                    "failed to remove corrupt database {}: {}",
541                    mmdb_path.display(),
542                    _remove_err
543                );
544            }
545            Err(anyhow::Error::new(e).context(format!(
546                "failed to open GeoLite2 database {}",
547                mmdb_path.display()
548            )))
549        }
550    }
551}

553/// Offline GeoIP lookup backed by cached GeoLite2 City and ASN databases.
554pub struct GeoLookup {
555    reader: Reader<Vec<u8>>,
556    asn_reader: Reader<Vec<u8>>,
557}
558
559impl GeoLookup {
560    /// Opens the GeoLite2 City and ASN databases, downloading them on first
561    /// use, so every lookup carries country/city/ASN data.
562    pub async fn new() -> anyhow::Result<Self> {
563        let mmdb_path = database_path()?;
564
565        let _download_guard = download_lock().lock().await;
566        let reader = ensure_database(
567            &mmdb_path,
568            GEOLITE_ENDPOINT_URL,
569            "the GeoLite2 city database",
570        )
571        .await?;
572        let asn_reader = ensure_database(
573            &asn_database_path()?,
574            GEOLITE_ASN_ENDPOINT_URL,
575            "the GeoLite2 ASN database",
576        )
577        .await?;
578
579        Ok(Self { reader, asn_reader })
580    }
581
582    /// Looks up country/city/ASN data for an IPv4 address.
583    ///
584    /// # Arguments
585    ///
586    /// * `ip` - Address to resolve; unknown IPs yield [`GeoData::default`].
587    pub fn lookup(&self, ip: &Ipv4Addr) -> GeoData {
588        let mut geodata = GeoData::default();
589        let Ok(result) = self.reader.lookup(std::net::IpAddr::V4(*ip)) else {
590            return geodata;
591        };
592        if let Ok(Some(city)) = result.decode::<City>() {
593            self.extract_country_data(&city, &mut geodata);
594            self.extract_continent_data(&city, &mut geodata);
595            self.extract_region_data(&city, &mut geodata);
596            self.extract_city_data(&city, &mut geodata);
597            self.extract_location_data(&city, &mut geodata);
598            self.extract_postal_data(&city, &mut geodata);
599        }
600        self.extract_ip_type(ip, &mut geodata);
601        geodata
602    }
603
604    fn extract_ip_type(&self, ip: &Ipv4Addr, geodata: &mut GeoData) {
605        let (asn, aso) = match self
606            .asn_reader
607            .lookup(std::net::IpAddr::V4(*ip))
608            .ok()
609            .and_then(|result| result.decode::<Asn>().ok())
610            .flatten()
611        {
612            Some(record) => (
613                record.autonomous_system_number,
614                record.autonomous_system_organization,
615            ),
616            None => (None, None),
617        };
618        geodata.asn = asn;
619        geodata.aso = aso.map(Box::from);
620        geodata.ip_type = IpType::classify(asn, aso, None, None);
621    }
622
623    fn extract_country_data(&self, lookup: &City, geodata: &mut GeoData) {
624        let country = &lookup.country;
625        if !country.is_empty() {
626            geodata.iso_code = country.iso_code.map(Box::from);
627            geodata.name = country.names.english.map(Box::from);
628        } else {
629            let continent = &lookup.continent;
630            geodata.iso_code = continent.code.map(Box::from);
631            geodata.name = continent.names.english.map(Box::from);
632        }
633    }
634
635    fn extract_region_data(&self, lookup: &City, geodata: &mut GeoData) {
636        if let Some(division) = lookup.subdivisions.first() {
637            geodata.region_iso_code = division.iso_code.map(Box::from);
638            geodata.region_name = division.names.english.map(Box::from);
639        }
640    }
641
642    fn extract_continent_data(&self, lookup: &City, geodata: &mut GeoData) {
643        geodata.continent_code = lookup.continent.code.map(Box::from);
644        geodata.continent_name = lookup.continent.names.english.map(Box::from);
645    }
646
647    fn extract_location_data(&self, lookup: &City, geodata: &mut GeoData) {
648        geodata.latitude = lookup.location.latitude;
649        geodata.longitude = lookup.location.longitude;
650        geodata.timezone = lookup.location.time_zone.map(Box::from);
