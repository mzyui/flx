1use std::{sync::Arc, time::Duration};

3/// Tuning knobs for provider scraping.
4pub struct Config {
5    /// Deduplicates candidates by endpoint and advertised protocols.
6    pub enforce_unique_ip: bool,
7    /// Caps concurrent provider fetches; must be non-zero.
8    pub concurrency_limit: usize,
9    /// Annotates candidates via GeoIP; required for country filters.
10    pub enable_geo_lookup: bool,
11    /// Allowlist of ISO country codes; empty means no filtering.
12    pub countries: Arc<[String]>,
13    /// Denylist of ISO country codes; empty means no filtering.
14    pub excluded_countries: Arc<[String]>,
15    /// Skips fallback providers once this many proxies exist.
16    pub fallback_threshold: Option<usize>,
17    /// Bounds the fallback phase; `None` waits for all sources.
18    pub fallback_phase_timeout: Option<Duration>,
19    /// Freshness window for cached source bodies; `None` disables the cache.
20    pub cache_ttl: Option<Duration>,
21    /// Re-fetches sources instead of serving cached bodies.
22    pub refresh_cache: bool,
23    /// Provider allowlist by name; empty selects all built-ins.
24    pub providers: Arc<[String]>,
25    /// Provider denylist by name.
26    pub excluded_providers: Arc<[String]>,
27    /// Extra raw source URLs scraped alongside built-ins.
28    pub custom_sources: Arc<[String]>,
29    /// Serves cached bodies only, opening no network connections.
30    pub offline: bool,
31    /// Minimum spacing between requests to the same host; `None` disables it.
32    pub fetch_delay: Option<Duration>,
33    /// Per-source fetch timeout override; `None` keeps each source default.
34    pub provider_timeout: Option<Duration>,
35}
36
37impl Config {
38    /// Returns the trimmed, uppercased country allowlist with empties dropped.
39    pub fn normalized_countries(&self) -> hashbrown::HashSet<String> {
40        self.countries
41            .iter()
42            .map(|country| country.trim().to_ascii_uppercase())
43            .filter(|country| !country.is_empty())
44            .collect()
45    }
46
47    /// Returns the trimmed, uppercased country denylist with empties dropped.
48    pub fn normalized_excluded_countries(&self) -> hashbrown::HashSet<String> {
49        self.excluded_countries
50            .iter()
51            .map(|country| country.trim().to_ascii_uppercase())
52            .filter(|country| !country.is_empty())
53            .collect()
54    }
55}
56
57/// Default concurrent provider fetches.
58pub const DEFAULT_CONCURRENCY_LIMIT: usize = 64;
59
60/// Default concurrent provider fetches per host.
61pub const DEFAULT_HOST_CONCURRENCY_LIMIT: usize = 4;
62
63/// Default source-cache freshness in minutes.
64pub const DEFAULT_CACHE_TTL_MINUTES: u64 = 15;
65
66impl Default for Config {
67    fn default() -> Self {
68        Self {
69            enforce_unique_ip: true,
70            concurrency_limit: DEFAULT_CONCURRENCY_LIMIT,
71            enable_geo_lookup: false,
72            countries: Arc::from(Vec::new()),
73            excluded_countries: Arc::from(Vec::new()),
74            fallback_threshold: None,
75            fallback_phase_timeout: None,
76            cache_ttl: Some(Duration::from_secs(
77                DEFAULT_CACHE_TTL_MINUTES.saturating_mul(60),
78            )),
79            refresh_cache: false,
80            providers: Arc::from(Vec::new()),
81            excluded_providers: Arc::from(Vec::new()),
82            custom_sources: Arc::from(Vec::new()),
83            offline: false,
84            fetch_delay: None,
85            provider_timeout: None,
86        }
87    }
88}

100    #[test]
101    fn country_filters_are_normalized_and_deduplicated() {
102        let config = Config {
103            countries: Arc::from(vec![
104                "id".to_owned(),
105                "ID".to_owned(),
106                " us ".to_owned(),
107                String::new(),
108            ]),
109            ..Config::default()
110        };
111
112        let countries = config.normalized_countries();
113
114        assert!(countries.contains("ID"));
115        assert!(countries.contains("US"));
116        assert_eq!(countries.len(), 2);
117    }
