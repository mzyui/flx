51pub(crate) const FETCH_CHANNEL_CAPACITY: usize = 2_048;

53/// Bound primary phase runtime before fallback decision.
54pub const PRIMARY_PHASE_TIMEOUT: Duration = Duration::from_secs(30);

56/// Start fallback providers after this long even while primary keeps running.
57const PRIMARY_FALLBACK_GRACE: Duration = Duration::from_secs(8);

59/// Stream scraped proxy candidates to consumers.
60///
61/// Build with [`ProxyFetcher::gather`]; the stream yields deduplicated
62/// proxies until providers are exhausted or `stop` fires.
63pub struct ProxyFetcher {
64    receiver: mpsc::Receiver<Proxy>,
65    counter: usize,
66    timer: time::Instant,
67    elapsed: Option<Duration>,
68    geolookup: Option<GeoLookup>,
69    countries: HashSet<String>,
70    excluded_countries: HashSet<String>,
71    unique_ips: DedupTable,
72    coordinator: JoinHandle<()>,
73    config: Config,
74    accepted: Arc<AtomicUsize>,
75    prefetched: Option<Proxy>,
76    stop_tx: watch::Sender<bool>,
77    stop_signaled: AtomicBool,
78    stages: Option<mpsc::Receiver<FetchStage>>,
79}

81impl ProxyFetcher {
82    /// Scrapes providers and streams deduplicated candidates.
83    ///
84    /// # Arguments
85    ///
86    /// * `config` - Providers, timeouts, cache, and GeoIP filters.
87    ///
88    /// # Errors
89    ///
90    /// Returns an error for invalid configs (zero concurrency or country
91    /// filter without GeoIP) or GeoIP init failures.
92    pub async fn gather(config: Config) -> anyhow::Result<Self> {
93        if config.concurrency_limit == 0 {
94            anyhow::bail!("config.concurrency_limit must be greater than zero");
95        }
96        // Reject country filter without GeoIP instead of dropping everything.
97        if (!config.countries.is_empty() || !config.excluded_countries.is_empty())
98            && !config.enable_geo_lookup
99        {
100            anyhow::bail!(
101                "country filter requires enable_geo_lookup=true (GeoIP lookup is currently disabled)"
102            );
103        }
104        let (sender, receiver) = mpsc::channel(FETCH_CHANNEL_CAPACITY);
105        let geolookup = if config.enable_geo_lookup {
106            Some(
107                GeoLookup::new()
108                    .await
109                    .context("failed to initialize geo lookup")?,
110            )
111        } else {
112            None
113        };
114
115        let registry = all_providers();
116        let mut known_names: Vec<&str> = registry.iter().map(|provider| provider.name()).collect();
117        let mut providers =
118            select_providers(registry, &config.providers, &config.excluded_providers);
119        if !config.custom_sources.is_empty() {
120            known_names.push("custom");
121        }
122        for name in config.providers.iter() {
123            if !known_names.contains(&name.as_str()) {
124                anyhow::bail!(
125                    "unknown provider `{name}` (available: {})",
126                    known_names.join(", ")
127                );
128            }
129        }
130        for url in config.custom_sources.iter() {
131            providers
132                .push(Arc::new(CustomUrlProvider::new(url).with_context(
133                    || format!("invalid custom source URL `{url}`"),
134                )?));
135        }
136        if providers.is_empty() {
137            #[cfg(feature = "log")]
138            log::warn!("provider selection matched no providers");
139        }
140
141        let countries = config.normalized_countries();
142        let excluded_countries = config.normalized_excluded_countries();
143        let accepted = Arc::new(AtomicUsize::new(0));
144
145        let mut primary = vec![];
146        let mut fallback = vec![];
147        for provider in providers.iter() {
148            let bucket = match provider.tier() {
149                ProviderTier::Primary => &mut primary,
150                ProviderTier::Fallback => &mut fallback,
151            };
152            for source in provider.sources() {
153                let source = match config.provider_timeout {
154                    Some(timeout) => source.with_timeout(timeout),
155                    None => source,
156                };
157                bucket.push((Arc::new(source), Arc::clone(provider)));
158            }
159        }
160
161        #[cfg(feature = "log")]
162        log::debug!(
163            "Proxy gathering started ({} primary sources, {} fallback sources)",
164            primary.len(),
165            fallback.len(),
166        );
167
168        let client = {
169            let mut http = HttpConnector::new();
170            // Cap TCP connect so dead hosts fail fast.
171            http.set_connect_timeout(Some(Duration::from_secs(6)));
172            http.enforce_http(false);
173            Arc::new(
174                Client::builder(TokioExecutor::new()).build::<_, Empty<Bytes>>(
175                    crate::proxy::client::https_connector_with_config(http, false),
176                ),
177            )
178        };
179        let concurrency_limit = config.concurrency_limit;
180        let fallback_threshold = config.fallback_threshold;
181        let fallback_phase_timeout = config.fallback_phase_timeout;
182        let offline = config.offline;
183        let fetch_delay = config.fetch_delay;
184
185        // Proceed without caching when the cache dir is unusable.
186        let fetch_cache = match config.cache_ttl {
187            Some(ttl) => match cache::Cache::new(ttl, config.refresh_cache) {
188                Ok(cache) => Some(Arc::new(cache)),
189                Err(error) => {
190                    #[cfg(feature = "log")]
191                    log::warn!("provider fetch cache disabled: {error:#}");
192                    let _ = error;
193                    None
194                }
195            },
196            None => None,
197        };
198
199        let throttle = Arc::new(Throttle::new());
200        let hosts = Arc::new(HostLimiter::new(DEFAULT_HOST_CONCURRENCY_LIMIT));
201
202        // Size primary phase from accepted count without extra channel hop.
203        let produced = Arc::clone(&accepted);
204
205        let (stop_tx, stop_rx) = watch::channel(false);
206        let stop_rx_coordinator = stop_rx.clone();
207        let stop_rx_primary = stop_rx.clone();
208        let stop_rx_fallback = stop_rx;
209
210        let (stage_tx, stages) = mpsc::channel(16);
211
212        // Run primary phase, then decide fallback; stop signal aborts early.
213        let coordinator = tokio::spawn({
214            let sender = sender.clone();
215            async move {
216                let stages = StageReporter { tx: stage_tx };
217                let settings = FetchSettings {
218                    fetch_cache,
219                    offline,
220                    throttle,
221                    fetch_delay,
222                    hosts,
223                };
224                let mut stop_rx_coordinator = stop_rx_coordinator;
225                let sem = Arc::new(Semaphore::new(concurrency_limit));
226                stages.send(FetchStage::Primary);
227                let primary_ctx = PhaseContext {
228                    client: Arc::clone(&client),
229                    sem: Arc::clone(&sem),
230                    tx: sender.clone(),
231                    stop_rx: stop_rx_primary,
232                    settings: settings.clone(),
233                };
234                let primary_started = time::Instant::now();
235                let mut primary_handles = spawn_phase(primary, &primary_ctx);
236
237                let mut primary_aborted = false;
238                tokio::select! {
239                    _ = async { while primary_handles.join_next().await.is_some() {} } => {
240                    }
241                    _ = stop_rx_coordinator.changed() => {
242                        primary_handles.abort_all();
243                        while primary_handles.join_next().await.is_some() {}
244                        primary_aborted = true;
245                    }
246                    _ = time::sleep(PRIMARY_FALLBACK_GRACE) => {}
247                }
248
249                #[cfg(feature = "log")]
250                if primary_aborted {
251                    log::warn!("primary providers aborted (stop signal)");
252                }
253                let _ = primary_aborted;
254
255                if *stop_rx_coordinator.borrow() {
256                    #[cfg(feature = "log")]
257                    log::debug!("stopping early: consumer collected enough proxies");
258                    primary_handles.abort_all();
259                    return;
260                }
261
262                let found = produced.load(Ordering::Relaxed);
263                if let Some(threshold) = fallback_threshold {
264                    if found >= threshold {
265                        #[cfg(feature = "log")]
266                        log::debug!(
267                            "skipping fallback providers: {} proxies already found (threshold {})",
268                            found,
269                            threshold
270                        );
271                        primary_handles.abort_all();
272                        return;
273                    }
274                }
275
276                #[cfg(feature = "log")]
277                log::debug!(
278                    "primary phase yielded {} proxies; starting {} fallback sources",
279                    found,
280                    fallback.len()
281                );
282
283                stages.send(FetchStage::Fallback);
284                let fallback_ctx = PhaseContext {
285                    client: Arc::clone(&client),
286                    sem: Arc::clone(&sem),
287                    tx: sender,
288                    stop_rx: stop_rx_fallback,
289                    settings: settings.clone(),
290                };
291                let mut fallback_handles = spawn_phase(fallback, &fallback_ctx);
292
293                let primary_remaining =
294                    PRIMARY_PHASE_TIMEOUT.saturating_sub(primary_started.elapsed());
295                if !primary_handles.is_empty() && !primary_remaining.is_zero() {
296                    tokio::select! {
297                        _ = async { while primary_handles.join_next().await.is_some() {} } => {}
298                        _ = time::sleep(primary_remaining) => {
299                            primary_handles.abort_all();
300                            while primary_handles.join_next().await.is_some() {}
301                            #[cfg(feature = "log")]
302                            log::warn!(
303                                "primary providers aborted after timeout {:?}",
304                                PRIMARY_PHASE_TIMEOUT
305                            );
306                        }
307                        _ = stop_rx_coordinator.changed() => {
308                            primary_handles.abort_all();
309                            fallback_handles.abort_all();
310                            while primary_handles.join_next().await.is_some() {}
311                            while fallback_handles.join_next().await.is_some() {}
312                            return;
313                        }
314                    }
315                }
316
317                match fallback_phase_timeout {
318                    Some(timeout) => {
319                        tokio::select! {
320                            _ = async { while fallback_handles.join_next().await.is_some() {} } => {}
321                            _ = time::sleep(timeout) => {
322                                fallback_handles.abort_all();
323                                while fallback_handles.join_next().await.is_some() {}
324                                #[cfg(feature = "log")]
325                                log::warn!(
326                                    "fallback providers aborted after timeout {:?}",
327                                    timeout
328                                );
329                            }
330                            _ = stop_rx_coordinator.changed() => {
331                                fallback_handles.abort_all();
332                                while fallback_handles.join_next().await.is_some() {}
333                            }
334                        }
335                    }
336                    None => {
337                        tokio::select! {
338                            _ = async { while fallback_handles.join_next().await.is_some() {} } => {}
339                            _ = stop_rx_coordinator.changed() => {
340                                fallback_handles.abort_all();
341                                while fallback_handles.join_next().await.is_some() {}
342                            }
343                        }
344                    }
345                }
346            }
347        });
348
349        Ok(Self {
350            receiver,
351            counter: 0,
352            timer: time::Instant::now(),
353            elapsed: None,
354            coordinator,
355            unique_ips: DedupTable::new(),
356            geolookup,
357            countries,
358            excluded_countries,
359            config,
360            accepted,
361            prefetched: None,
362            stop_tx,
363            stop_signaled: AtomicBool::new(false),
364            stages: Some(stages),
365        })
366    }
367}

369#[cfg(test)]
370async fn finish_phase(mut handles: tokio::task::JoinSet<()>, timeout: Duration) -> bool {
371    let drain = async { while handles.join_next().await.is_some() {} };
372    if time::timeout(timeout, drain).await.is_ok() {
373        return true;
374    }
375    handles.abort_all();
376    while handles.join_next().await.is_some() {}
377    false
378}

380impl ProxyFetcher {
381    /// Returns the next accepted candidate, or `None` when exhausted.
382    ///
383    /// Skips duplicates and country-filtered proxies; `None` means all
384    /// providers finished and the channel closed.
385    pub async fn get_one(&mut self) -> Option<Proxy> {
386        loop {
387            let proxy = if let Some(proxy) = self.prefetched.take() {
388                proxy
389            } else {
390                match self.receiver.recv().await {
391                    Some(proxy) => proxy,
392                    None => {
393                        self.elapsed = Some(self.timer.elapsed());
394                        return None;
395                    }
396                }
397            };
398            if let Some(proxy) = self.accept(proxy) {
399                self.check_drain();
400                return Some(proxy);
401            }
402        }
403    }
404
405    /// Share accepted-proxy count with consumers.
406    pub fn accepted_handle(&self) -> Arc<AtomicUsize> {
407        Arc::clone(&self.accepted)
408    }
409
410    /// Take gathering-phase events for this run.
411    pub fn stage_events(&mut self) -> Option<mpsc::Receiver<FetchStage>> {
412        self.stages.take()
413    }
414
415    fn check_drain(&mut self) {
416        if let Ok(proxy) = self.receiver.try_recv() {
417            self.prefetched = Some(proxy);
418        }
419    }
420
421    fn accept(&mut self, proxy: Proxy) -> Option<Proxy> {
422        let mut ctx = AcceptContext {
423            unique_ips: &mut self.unique_ips,
424            enforce_unique_ip: self.config.enforce_unique_ip,
425            countries: &self.countries,
426            excluded_countries: &self.excluded_countries,
427            counter: &mut self.counter,
428            accepted: &self.accepted,
429        };
430        let result = accept_proxy(&mut ctx, proxy, |ip| {
431            self.geolookup.as_ref().map(|geolookup| {
432                // Move mmdb reads off async workers; read directly inline otherwise.
433                let do_lookup = || geolookup.lookup(ip);
434                match tokio::runtime::Handle::try_current() {
435                    Ok(handle)
436                        if handle.runtime_flavor()
437                            == tokio::runtime::RuntimeFlavor::MultiThread =>
438                    {
439                        tokio::task::block_in_place(do_lookup)
440                    }
441                    _ => do_lookup(),
442                }
443            })
444        });
445        if result.is_some() {
446            if let Some(threshold) = self.config.fallback_threshold {
447                // Bump stop version exactly once via atomic compare-exchange.
448                if self.accepted.load(Ordering::Relaxed) >= threshold
449                    && self
450                        .stop_signaled
451                        .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
452                        .is_ok()
453                {
454                    let _ = self.stop_tx.send(true);
455                }
456            }
457        }
458        result
459    }
460}

462struct AcceptContext<'a> {
463    unique_ips: &'a mut DedupTable,
464    enforce_unique_ip: bool,
465    countries: &'a HashSet<String>,
466    excluded_countries: &'a HashSet<String>,
467    counter: &'a mut usize,
468    accepted: &'a AtomicUsize,
469}
470
471fn accept_proxy<F>(ctx: &mut AcceptContext, mut proxy: Proxy, lookup: F) -> Option<Proxy>
472where
473    F: FnOnce(&Ipv4Addr) -> Option<crate::geolookup::models::GeoData>,
474{
475    if ctx.enforce_unique_ip
476        && !ctx
477            .unique_ips
478            .insert((proxy.ip, proxy.port, protocol_hash(&proxy.expected_types)))
479    {
480        return None;
481    }
482
483    if let Some(geo) = lookup(&proxy.ip) {
484        proxy.geo = Arc::new(geo);
485
486        if !ctx.countries.is_empty()
487            && !proxy
488                .geo
489                .iso_code
490                .as_ref()
491                .map(|code| ctx.countries.contains(code.as_ref()))
492                .unwrap_or(false)
493        {
494            return None;
495        }
496
497        if !ctx.excluded_countries.is_empty()
498            && proxy
499                .geo
500                .iso_code
501                .as_ref()
502                .map(|code| ctx.excluded_countries.contains(code.as_ref()))
503                .unwrap_or(false)
504        {
505            return None;
506        }
507    }
508
509    *ctx.counter += 1;
510    ctx.accepted.fetch_add(1, Ordering::Relaxed);
511    Some(proxy)
512}

514impl Stream for ProxyFetcher {
515    type Item = Proxy;
516
517    fn poll_next(self: Pin<&mut Self>, cx: &mut TaskContext<'_>) -> Poll<Option<Self::Item>> {
518        let this = self.get_mut();
519        loop {
520            if let Some(proxy) = this.prefetched.take() {
521                if let Some(proxy) = this.accept(proxy) {
522                    this.check_drain();
523                    return Poll::Ready(Some(proxy));
524                }
525                continue;
526            }
527            match this.receiver.poll_recv(cx) {
528                Poll::Ready(Some(proxy)) => {
529                    if let Some(proxy) = this.accept(proxy) {
530                        this.check_drain();
531                        return Poll::Ready(Some(proxy));
532                    }
533                }
534                Poll::Ready(None) => {
535                    if this.elapsed.is_none() {
536                        this.elapsed = Some(this.timer.elapsed());
537                    }
538                    return Poll::Ready(None);
539                }
540                Poll::Pending => return Poll::Pending,
541            }
542        }
543    }
544}

546impl Drop for ProxyFetcher {
547    fn drop(&mut self) {
548        // Fail pending sends to unwind producers before aborting coordinator.
549        self.receiver.close();
550
551        self.coordinator.abort();
552
553        #[cfg(feature = "log")]
554        log::debug!(
555            "Proxy gathering completed: {} proxies found ({:?})",
556            self.counter,
557            self.elapsed.unwrap_or(self.timer.elapsed()),
558        );
559    }
560}
